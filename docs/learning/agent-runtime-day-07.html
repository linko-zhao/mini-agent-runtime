单包通过无法证明装配时真的传入了超时和 Sink；最终回答也可能掩盖工具根本没有执行。需要同时观察调用次数、历史与终态。

先解释概念，再看一次具体运行

宿主装配

是什么：把具体依赖连接到已完成的 Runtime。

为什么需要：工具列表、模型、步数、超时和 Sink 一次交给 BuildWeek01。

落到代码：BuildWeek01 只验证列表并调用 Agent.New。

集成矩阵

是什么：让一组输入覆盖不同组件组合。

为什么需要：正常、未知工具、错误、取消、超时和预算耗尽。

落到代码：每个场景同时检查输出、模型轮数、执行数和唯一终态。

可观察证据

是什么：行为断言，而非最终文本看起来合理。

为什么需要：正常读取必须在第二轮历史看到 read-1 的正文。

落到代码：脚本模型自己断言 Role、ID 与 Content。

失败反例

maxSteps=1 时工具确实执行一次，但 Run 返回 ErrMaxSteps，发布唯一 run_failed。它不能被标成正常回答，也不能声称工具副作用已回滚。

为什么这样设计

Week01Config

职责与字段/参数：Model 必须非 nil；Tools 是工具列表，可 nil/空；MaxSteps 是正模型轮次数；ToolTimeout 为 time.Duration，零禁用、负值拒绝；Sink 可 nil。

返回与失败：配置由宿主提供，Build 不执行模型或工具；列表转入独立 map，工具对象仍共享，之后不得并发改依赖。

状态、所有权与取消：不提供热更新、关闭管理或并发 Run 承诺。

设计理由：用列表保留重复名信息，避免 map 字面量先覆盖导致无法检测。

BuildWeek01

职责与字段/参数：BuildWeek01(c) 返回 *agent.Agent,error；先检查 nil 工具和重复 Name，再交 Agent.New 验证其余配置并做绑定。

返回与失败：任何失败返回 nil,error，无部分 Agent；依赖原因链由 New 保留。空名由 New 拒绝；typed nil 按前序约定不检查。

状态、所有权与取消：成功只装配，不 Generate/Execute；WithToolTimeout/WithEventSink 必须完整传递。

设计理由：把宿主接线错误与 Loop 算法错误分开定位。

复用的运行契约

职责与字段/参数：Run、Model、Tool、EventSink 不改签名。七场景分别明确成功/失败，取消和超时保持 errors.Is。

返回与失败：正常两轮检查三条历史；未知工具不执行；预先取消不调用模型；超时返回后不再进入模型。

状态、所有权与取消：timeout fake 等待 Done，不靠 sleep；脚本每个子场景独立，不使用共享游标。

设计理由：同时看计数、关联和终态，才能发现装配缺失。

行为契约：不用读答案也知道怎么写

按编号实现；未规定的错误文本不需逐字匹配。依赖错误用 errors.Is/As 判断，不以私有字段或某种容器为验收目标。

type script struct {
	round int
	mode  string
	t     *testing.T
}

func (m *script) Generate(c context.Context, ms []model.Message) (*model.Response, error) {
	m.round++
	if m.mode == "model_error" {
		return nil, errModel
	}
	if m.mode == "text" || m.round == 2 {
		if m.round == 2 && (len(ms) != 3 || ms[2].ToolCallID != "read-1" || ms[2].Content != "正文") {
			m.t.Fatal(ms)
		}
		return &model.Response{Message: model.Message{Role: "assistant", Content: "已读取"}}, nil
	}
	name := "get_document"
	if m.mode == "unknown" {
		name = "missing"
	}
	return &model.Response{Message: model.Message{Role: "assistant", ToolCalls: []model.ToolCall{{ID: "read-1", Name: name, Arguments: json.RawMessage(`{"id":"doc-1"}`)}}}}, nil
}

var errModel = errors.New("model failed")

type reader struct {
	calls   int
	timeout bool
}

func (*reader) Name() string { return "get_document" }
func (r *reader) Execute(c context.Context, a json.RawMessage) (string, error) {
	r.calls++
	if string(a) != `{"id":"doc-1"}` {
		return "", errors.New("args")
	}
	if r.timeout {
		<-c.Done()
		return "late", nil
	}
	return "正文", nil
}

type sink struct{ es []agent.Event }

func (s *sink) Emit(_ context.Context, e agent.Event) { s.es = append(s.es, e) }
func TestDay07Matrix(t *testing.T) {
	for _, tc := range []struct {
		mode               string
		max, reads, rounds int
		want               error
		timeout, cancel    bool
	}{
		{"text", 2, 0, 1, nil, false, false}, {"roundtrip", 2, 1, 2, nil, false, false}, {"unknown", 2, 0, 1, nil, false, false}, {"model_error", 2, 0, 1, errModel, false, false}, {"cancel", 2, 0, 0, context.Canceled, false, true}, {"timeout", 2, 1, 1, context.DeadlineExceeded, true, false}, {"max", 1, 1, 1, agent.ErrMaxSteps, false, false},
	} {
		t.Run(tc.mode, func(t *testing.T) {
			m := &script{mode: tc.mode, t: t}
			r := &reader{timeout: tc.timeout}
			s := &sink{}
			cfg := assembly.Week01Config{Model: m, Tools: []tool.Tool{r}, MaxSteps: tc.max, Sink: s}
			if tc.timeout {
				cfg.ToolTimeout = time.Nanosecond
			}
			a, e := assembly.BuildWeek01(cfg)
			if e != nil {
				t.Fatal(e)
			}
			ctx, cancel := context.WithCancel(context.Background())
			defer cancel()
			if tc.cancel {
				cancel()
			}
			out, e := a.Run(ctx, "读取 doc-1")
			failed := tc.want != nil || tc.mode == "unknown"
			if failed {
				if e == nil || out != "" {
					t.Fatal(out, e)
				}
				if tc.want != nil && !errors.Is(e, tc.want) {
					t.Fatal(e)
				}
			} else if e != nil || out != "已读取" {
				t.Fatal(out, e)
			}
			if r.calls != tc.reads || m.round != tc.rounds {
				t.Fatal(r.calls, m.round)
			}
			terminal := 0
			for _, ev := range s.es {
				if ev.Type == agent.EventRunFailed || ev.Type == agent.EventRunFinished {
					terminal++
					if (ev.Type == agent.EventRunFailed) != failed {
						t.Fatal(ev)
					}
				}
			}
			if terminal != 1 {
				t.Fatal(s.es)
			}
		})
	}
}

package assembly

import (
	"fmt"
	"github.com/linko-zhao/mini-agent-runtime/internal/agent"
	"github.com/linko-zhao/mini-agent-runtime/internal/model"
	"github.com/linko-zhao/mini-agent-runtime/internal/tool"
	"time"
)

type Week01Config struct {
	Model       model.Model
	Tools       []tool.Tool
	MaxSteps    int
	ToolTimeout time.Duration
	Sink        agent.EventSink
}

func BuildWeek01(c Week01Config) (*agent.Agent, error) {
	tools := map[string]tool.Tool{}
	for _, t := range c.Tools {
		if t == nil {
			return nil, fmt.Errorf("nil tool")
		}
		n := t.Name()
		if _, ok := tools[n]; ok {
			return nil, fmt.Errorf("duplicate tool")
		}
		tools[n] = t
	}
	return agent.New(c.Model, tools, c.MaxSteps, agent.WithToolTimeout(c.ToolTimeout), agent.WithEventSink(c.Sink))
}
