今天只解决什么

今天实现同步执行事件，让宿主能观察读取过程的顺序、耗时和结束状态。

Run 只返回最后文本，无法判断时间花在模型还是工具。到处打印原始参数既难以验证顺序，也会让正文混进运行日志。

只发布内存同步事件；不实现日志平台、追踪持久化、实时查询、重入或异常隔离。共享 Sink 的串行保护放到 Day06。

先解释概念，再看一次具体运行

事件

是什么：对已经发生的阶段作简短记录。

为什么需要：tool_completed 说明 read-1 返回，不携带正文。

落到代码：Event 包含类型、轮次、时间和状态。

接收器

是什么：由宿主提供的观察目标。

为什么需要：测试把事件记到切片中检查顺序。

落到代码：EventSink.Emit(ctx,event) 同步调用。

唯一终态

是什么：每次 Run 最后只发布一次结束或失败。

为什么需要：超时只产生 run_failed，不再发 run_finished。

落到代码：Run 的 defer 根据命名返回错误选择事件。

失败反例

模型报错时记录 model_completed(status=error) 后发布唯一 run_failed。Run 预先取消仍发 started/failed，但不发 step_started，也不调用模型。

为什么这样设计

EventType / 常量

职责与字段/参数：run_started、step_started、model_completed、tool_started、tool_completed、run_finished、run_failed 是稳定字符串。支持文件中预留 tool_authorized/tool_denied 供后续课程，今天不发布。

返回与失败：事件顺序见运行图；只有实际开始的调用才有 completed。nil 响应视为模型错误，非 assistant 由 Loop 检查后失败。

状态、所有权与取消：常量无状态，不承诺真实模型语义正确。

设计理由：用稳定常量替代散落文本。

Event

职责与字段/参数：Type 为阶段；Step 是从1开始的模型轮次，run_started 为0；At 由 emit 写当前时间；Duration 是 time.Duration，完成/终态非负，started 默认0。

返回与失败：ToolName/ToolCallID 仅工具事件有值；Status 为 ok/error/cancelled。PrincipalID 是后续授权预留，本日空。事件不含 Prompt、参数、正文、原错误。

状态、所有权与取消：Event 按值传递；终态步数为最后实际轮次，预算耗尽时为 maxSteps。时间仅观测，不断言固定耗时。

设计理由：分离运行信息与业务内容，降低意外泄露。

EventSink.Emit / WithEventSink

职责与字段/参数：Emit(ctx,Event) 不返回错误；WithEventSink(s) 返回构造选项，s=nil 禁用观察。emit 为事件填 At 后同步转发。

返回与失败：Sink 返回才继续业务；不吞 panic、不重试、不保证持久保存。

状态、所有权与取消：取消时仍把原 ctx 交给 Sink，以便记录终态；Sink 应快速返回、不重入同一 Agent。今天不承诺并发安全。

设计理由：简单接口便于 fake；同步的代价是观察者会影响延迟。

Run / status

职责与字段/参数：Run 使用命名返回值，让 defer 依据 runErr 只发布一个终态。status 通过 errors.Is 区分取消/超时，其他错误为 error，nil 为 ok。

返回与失败：普通回答输出不变；失败仍保留原错误；started 以后所有普通返回路径都有一个终态，panic 不在契约范围。

状态、所有权与取消：复用 Day04 的 context、超时和计时器释放。sink=nil 时不改变执行结果。

设计理由：统一收口，避免每个错误出口重复发终态。

行为契约：不用读答案也知道怎么写

按编号实现；未规定的错误文本不需逐字匹配。依赖错误用 errors.Is/As 判断，不以私有字段或某种容器为验收目标。

package agent_test

import (
	"context"
	"encoding/json"
	"errors"
	"github.com/linko-zhao/mini-agent-runtime/internal/agent"
	"github.com/linko-zhao/mini-agent-runtime/internal/model"
	"github.com/linko-zhao/mini-agent-runtime/internal/tool"
	"reflect"
	"testing"
	"time"
)

type events struct{ all []agent.Event }

func (s *events) Emit(_ context.Context, e agent.Event) { s.all = append(s.all, e) }

type day05Model func(context.Context, []model.Message) (*model.Response, error)

func (f day05Model) Generate(c context.Context, m []model.Message) (*model.Response, error) {
	return f(c, m)
}

type day05Tool struct {
	f func(context.Context, json.RawMessage) (string, error)
}

func (day05Tool) Name() string                                                   { return "get_document" }
func (f day05Tool) Execute(c context.Context, a json.RawMessage) (string, error) { return f.f(c, a) }
func day05Answer(s string) *model.Response {
	return &model.Response{Message: model.Message{Role: "assistant", Content: s}}
}
func day05Request() *model.Response {
	return &model.Response{Message: model.Message{Role: "assistant", ToolCalls: []model.ToolCall{{ID: "read-1", Name: "get_document", Arguments: json.RawMessage(`{"id":"doc-1"}`)}}}}
}
func day05Must(t *testing.T, m model.Model, tools map[string]tool.Tool, n int) *agent.Agent {
	t.Helper()
	a, e := agent.New(m, tools, n)
	if e != nil {
		t.Fatal(e)
	}
	return a
}

func TestDay05Sequence(t *testing.T) {
	s := &events{}
	round := 0
	m := day05Model(func(context.Context, []model.Message) (*model.Response, error) {
		round++
		if round == 1 {
			return day05Request(), nil
		}
		return day05Answer("已读取"), nil
	})
	a, e := agent.New(m, map[string]tool.Tool{"get_document": day05Tool{func(context.Context, json.RawMessage) (string, error) { return "正文", nil }}}, 2, agent.WithEventSink(s))
	if e != nil {
		t.Fatal(e)
	}
	if _, e = a.Run(context.Background(), "读取 doc-1"); e != nil {
		t.Fatal(e)
	}
	want := []agent.EventType{agent.EventRunStarted, agent.EventStepStarted, agent.EventModelCompleted, agent.EventToolStarted, agent.EventToolCompleted, agent.EventStepStarted, agent.EventModelCompleted, agent.EventRunFinished}
	var got []agent.EventType
	for _, ev := range s.all {
		got = append(got, ev.Type)
		if ev.At.IsZero() || ev.Duration < 0 || ev.Status != "ok" {
			t.Fatal(ev)
		}
	}
	if !reflect.DeepEqual(got, want) {
		t.Fatal(got)
	}
	for _, i := range []int{3, 4} {
		ev := s.all[i]
		if ev.Step != 1 || ev.ToolName != "get_document" || ev.ToolCallID != "read-1" {
			t.Fatal(ev)
		}
	}
	if s.all[len(s.all)-1].Step != 2 {
		t.Fatal("terminal step")
	}
}
func TestDay05Terminal(t *testing.T) {
	sentinel := errors.New("provider")
	for _, tc := range []struct {
		name   string
		m      day05Model
		cancel bool
		want   string
	}{
		{"error", func(context.Context, []model.Message) (*model.Response, error) { return nil, sentinel }, false, "error"},
		{"nil", func(context.Context, []model.Message) (*model.Response, error) { return nil, nil }, false, "error"},
		{"cancelled", func(context.Context, []model.Message) (*model.Response, error) {
			t.Fatal("day05Must not call")
			return nil, nil
		}, true, "cancelled"},
	} {
		t.Run(tc.name, func(t *testing.T) {
			s := &events{}
			a, e := agent.New(tc.m, nil, 1, agent.WithEventSink(s))
			if e != nil {
				t.Fatal(e)
			}
			ctx, cancel := context.WithCancel(context.Background())
			defer cancel()
			if tc.cancel {
				cancel()
			}
			if _, e = a.Run(ctx, "secret prompt"); e == nil {
				t.Fatal("want failure")
			}
			n := 0
			for _, v := range s.all {
				if v.Type == agent.EventRunFinished || v.Type == agent.EventRunFailed {
					n++
					if v.Type != agent.EventRunFailed || v.Status != tc.want {
						t.Fatal(v)
					}
				}
			}
			if n != 1 {
				t.Fatal(s.all)
			}
		})
	}
	// A nil sink disables observation without changing results.
	a, e := agent.New(day05Model(func(context.Context, []model.Message) (*model.Response, error) { return day05Answer("ok"), nil }), nil, 1, agent.WithEventSink(nil))
	if e != nil {
		t.Fatal(e)
	}
	if s, e := a.Run(context.Background(), ""); e != nil || s != "ok" {
		t.Fatal(s, e)
	}
}

func TestDay05ToolFailuresAndBudget(t *testing.T) {
	for _, tc := range []struct {
		name    string
		status  string
		timeout bool
		max     int
		want    error
	}{
		{"tool_error", "error", false, 2, errors.New("tool failure")},
		{"timeout", "cancelled", true, 2, context.DeadlineExceeded},
		{"budget", "error", false, 1, agent.ErrMaxSteps},
	} {
		t.Run(tc.name, func(t *testing.T) {
			s := &events{}
			m := day05Model(func(context.Context, []model.Message) (*model.Response, error) { return day05Request(), nil })
			r := day05Tool{func(c context.Context, _ json.RawMessage) (string, error) {
				if tc.timeout {
					<-c.Done()
					return "late", nil
				}
				if tc.name == "tool_error" {
					return "", tc.want
				}
				return "正文", nil
			}}
			a, e := agent.New(m, map[string]tool.Tool{"get_document": r}, tc.max, agent.WithEventSink(s), agent.WithToolTimeout(func() time.Duration {
				if tc.timeout {
					return time.Nanosecond
				}
				return 0
			}()))
			if e != nil {
				t.Fatal(e)
			}
			_, e = a.Run(context.Background(), "读取 doc-1")
			if !errors.Is(e, tc.want) {
				t.Fatal(e)
			}
			completed, terminal := 0, 0
			for _, ev := range s.all {
				if ev.Type == agent.EventToolCompleted {
					completed++
					want := "ok"
					if tc.name != "budget" {
						want = tc.status
					}
					if ev.Status != want {
						t.Fatal(ev)
					}
				}
				if ev.Type == agent.EventRunFailed || ev.Type == agent.EventRunFinished {
					terminal++
					if ev.Type != agent.EventRunFailed || ev.Status != tc.status {
						t.Fatal(ev)
					}
				}
			}
			if completed != 1 || terminal != 1 {
				t.Fatal(s.all)
			}
		})
	}
}

package agent

import (
	"context"
	"errors"
	"fmt"
	"github.com/linko-zhao/mini-agent-runtime/internal/model"
	"time"
)

func (a *Agent) Run(ctx context.Context, input string) (output string, runErr error) {
	started := time.Now()
	step := 0
	a.emit(ctx, Event{Type: EventRunStarted, Status: "ok"})
	defer func() {
		kind := EventRunFinished
		if runErr != nil {
			kind = EventRunFailed
		}
		a.emit(ctx, Event{Type: kind, Step: step, Duration: time.Since(started), Status: status(runErr)})
	}()
	messages := []model.Message{{Role: "user", Content: input}}
	for step = 1; step <= a.maxSteps; step++ {
		if e := ctx.Err(); e != nil {
			return "", e
		}
		a.emit(ctx, Event{Type: EventStepStarted, Step: step, Status: "ok"})
		modelStart := time.Now()
		r, e := a.model.Generate(ctx, messages)
		if ctx.Err() != nil {
			e = ctx.Err()
		}
		if e == nil && r == nil {
			e = fmt.Errorf("nil response")
		}
		a.emit(ctx, Event{Type: EventModelCompleted, Step: step, Duration: time.Since(modelStart), Status: status(e)})
		if e != nil {
			return "", e
		}
		if r == nil || r.Message.Role != "assistant" {
			return "", fmt.Errorf("agent: invalid response")
		}
		messages = append(messages, r.Message)
		if len(r.Message.ToolCalls) == 0 {
			return r.Message.Content, nil
		}
		for _, call := range r.Message.ToolCalls {
			if e := ctx.Err(); e != nil {
				return "", e
			}
			t, ok := a.tools[call.Name]
			if !ok {
				return "", fmt.Errorf("agent: unknown tool")
			}
			tc := ctx
			cancel := func() {}
			if a.toolTimeout > 0 {
				tc, cancel = context.WithTimeout(ctx, a.toolTimeout)
			}
			start := time.Now()
			ev := Event{Type: EventToolStarted, Step: step, ToolName: call.Name, ToolCallID: call.ID, Status: "ok"}
			a.emit(ctx, ev)
			out, e := t.Execute(tc, call.Arguments)
			if tc.Err() != nil {
				e = tc.Err()
			}
			cancel()
			ev.Type = EventToolCompleted
			ev.Duration = time.Since(start)
			ev.Status = status(e)
			a.emit(ctx, ev)
			if e != nil {
				return "", e
			}
			messages = append(messages, model.Message{Role: "tool", Content: out, ToolCallID: call.ID})
		}
	}
	step = a.maxSteps
	return "", ErrMaxSteps
}

func status(e error) string {
	if errors.Is(e, context.Canceled) || errors.Is(e, context.DeadlineExceeded) {
		return "cancelled"
	}
	if e != nil {
		return "error"
	}
	return "ok"
}

package agent

import (
	"context"
	"time"
)

type EventType string

const (
	EventRunStarted     EventType = "run_started"
	EventStepStarted    EventType = "step_started"
	EventModelCompleted EventType = "model_completed"
	EventToolStarted    EventType = "tool_started"
	EventToolAuthorized EventType = "tool_authorized"
	EventToolDenied     EventType = "tool_denied"
	EventToolCompleted  EventType = "tool_completed"
	EventRunFinished    EventType = "run_finished"
	EventRunFailed      EventType = "run_failed"
)

type Event struct {
	Type        EventType
	Step        int
	At          time.Time
	ToolName    string
	ToolCallID  string
	PrincipalID string // trusted opaque ID on authorization decision events only
	Duration    time.Duration
	Status      string
}
type EventSink interface{ Emit(context.Context, Event) }

func (a *Agent) emit(ctx context.Context, e Event) {
	if a.sink != nil {
		e.At = time.Now()
		a.sink.Emit(ctx, e)
	}
}
