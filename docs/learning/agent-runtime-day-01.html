今天只解决什么

今天实现最小 Agent Loop，让“读取 doc-1”的请求能够经历模型 → 工具 → 模型并返回回答。

模型只会提出调用请求，不会替宿主执行工具。如果把业务读取直接写进模型适配器，换一个资料库就必须修改协议代码。

本日串行执行，不处理工具说明绑定、单工具超时、事件、重试或并发。工具错误暂时终止 Run；Day09 再迁移为受控错误回填。

先解释概念，再看一次具体运行

模型

是什么：根据当前消息给出下一条 assistant 消息。

为什么需要：先给出 get_document 请求，读到正文后再给“已读取”。

落到代码：Model.Generate 接收历史，返回 Response。

工具

是什么：真正访问业务数据的对象。

为什么需要：get_document 接收 {"id":"doc-1"}，返回“正文”。

落到代码：Tool.Execute 接收本次 context 和 JSON 参数。

运行循环

是什么：决定下一步调用谁、何时结束的控制层。

为什么需要：先问模型，再执行工具，最后把结果放回历史。

落到代码：Agent.Run 私有保存 messages 和 step。

失败反例

若 get_document 未注册，Run 返回错误，不执行任何工具；若 maxSteps=1，第一次工具结果已得到，但没有预算进入第二轮，返回 ErrMaxSteps。

为什么这样设计

Message / ToolCall / Response

职责与字段/参数：Message.Role 是 user、assistant 或 tool；Content 是正文，可空。ToolCalls 是有序请求；ToolCallID 仅在结果中引用请求 ID。ToolCall.ID 标识一次调用，Name 路由工具，Arguments 是 JSON 字节；Response.Message 是本轮模型产物。

返回与失败：这些值由调用方提供，Loop 只追加自己的历史；当天不承诺对不可信依赖深拷贝。

状态、所有权与取消：无网络、取消或内部锁；不要让模型/工具修改传入切片。

设计理由：把协议与执行分开，同名工具可以有多次不同调用。

Model.Generate / Tool.Name / Tool.Execute

职责与字段/参数：Generate(ctx,messages) 返回非 nil assistant Response 或 error；Execute(ctx,args) 返回文本或 error；Name 返回稳定注册名。

返回与失败：不返回可继续消费的部分结果；依赖错误原样保留供 errors.Is 检查。

状态、所有权与取消：调用同步；ctx 不可 nil，下游负责合作取消。工具副作用不会被 Loop 回滚。

设计理由：测试可注入脚本模型和内存工具，无需真实 API。

Agent / New

职责与字段/参数：New(m,tools,maxSteps) 要求 m 非 nil、maxSteps>0（单位：模型轮次）。tools 可 nil；每项非 nil、key 非空且等于 Name；构造复制 map。

返回与失败：成功返回 *Agent；非法配置返回 nil,error；构造不执行工具或模型。

状态、所有权与取消：工具对象仍共享；接口中的 typed nil 不在本日检查范围。map 组装前的重复 key 已被 Go 覆盖，Day07 在列表入口拒绝重名。

设计理由：保护注册集合，不能声称复制了业务对象。

Run / ErrMaxSteps

职责与字段/参数：Run(ctx,input) 接受任意字符串（允许空）；成功返回最后一条无 ToolCalls 的 assistant.Content。每次 Run 新建历史；无结果时返回空文本,error。

返回与失败：nil 响应、非 assistant、未知工具终止；模型/工具错误保留链；预算耗尽 errors.Is(err,ErrMaxSteps)。

状态、所有权与取消：每轮模型前检查 ctx；工具收到同一个 ctx。执行后取消优先与更细边界留到 Day04；今天无共享并发承诺。

设计理由：有限循环避免脚本永远要求工具。

Mock / NewMock / Generate

职责与字段/参数：NewMock(responses) 接收有序脚本，nil 等于空脚本；Generate 不根据历史推理，依序返回响应副本。

返回与失败：耗尽返回 nil,error；预先取消返回 ctx.Err，且不推进游标。

状态、所有权与取消：脚本切片由测试持有，构造后不要修改；返回值是结构体浅拷贝。每个 Run 创建一个 Mock，不共享并发。

设计理由：它证明控制流与数据关联，不能证明真实模型会选对工具。

行为契约：不用读答案也知道怎么写

按编号实现；未规定的错误文本不需逐字匹配。依赖错误用 errors.Is/As 判断，不以私有字段或某种容器为验收目标。

package agent_test

import (
	"context"
	"encoding/json"
	"errors"
	"github.com/linko-zhao/mini-agent-runtime/internal/agent"
	"github.com/linko-zhao/mini-agent-runtime/internal/model"
	"github.com/linko-zhao/mini-agent-runtime/internal/tool"
	"testing"
)

type dModel func(context.Context, []model.Message) (*model.Response, error)

func (f dModel) Generate(c context.Context, m []model.Message) (*model.Response, error) {
	return f(c, m)
}

type dTool struct {
	f func(context.Context, json.RawMessage) (string, error)
}

func (dTool) Name() string                                                   { return "get_document" }
func (f dTool) Execute(c context.Context, a json.RawMessage) (string, error) { return f.f(c, a) }
func answer(s string) *model.Response {
	return &model.Response{Message: model.Message{Role: "assistant", Content: s}}
}
func request() *model.Response {
	return &model.Response{Message: model.Message{Role: "assistant", ToolCalls: []model.ToolCall{{ID: "read-1", Name: "get_document", Arguments: json.RawMessage(`{"id":"doc-1"}`)}}}}
}
func must(t *testing.T, m model.Model, tools map[string]tool.Tool, n int) *agent.Agent {
	t.Helper()
	a, e := agent.New(m, tools, n)
	if e != nil {
		t.Fatal(e)
	}
	return a
}

func TestDay01RoundTrip(t *testing.T) {
	calls, reads := 0, 0
	m := dModel(func(c context.Context, ms []model.Message) (*model.Response, error) {
		calls++
		if calls == 1 {
			if len(ms) != 1 || ms[0].Role != "user" || ms[0].Content != "读取 doc-1" {
				t.Fatal(ms)
			}
			return request(), nil
		}
		if len(ms) != 3 || ms[1].Role != "assistant" || ms[2].Role != "tool" || ms[2].ToolCallID != "read-1" || ms[2].Content != "正文" {
			t.Fatal(ms)
		}
		return answer("已读取"), nil
	})
	tools := map[string]tool.Tool{"get_document": dTool{func(c context.Context, a json.RawMessage) (string, error) {
		reads++
		if string(a) != `{"id":"doc-1"}` {
			t.Fatal(string(a))
		}
		return "正文", nil
	}}}
	a := must(t, m, tools, 2)
	delete(tools, "get_document")
	out, e := a.Run(context.Background(), "读取 doc-1")
	if e != nil || out != "已读取" || calls != 2 || reads != 1 {
		t.Fatalf("%q %v calls=%d reads=%d", out, e, calls, reads)
	}
}
func TestDay01Boundaries(t *testing.T) {
	m := dModel(func(context.Context, []model.Message) (*model.Response, error) { return answer("完成"), nil })
	for _, n := range []int{0, -1} {
		if _, e := agent.New(m, nil, n); e == nil {
			t.Fatal("maxSteps")
		}
	}
	if _, e := agent.New(nil, nil, 1); e == nil {
		t.Fatal("nil model")
	}
	if _, e := agent.New(m, map[string]tool.Tool{"bad": dTool{}}, 1); e == nil {
		t.Fatal("name mismatch")
	}
	if _, e := agent.New(m, map[string]tool.Tool{"get_document": nil}, 1); e == nil {
		t.Fatal("nil tool")
	}
	for _, tc := range []struct {
		name string
		f    dModel
	}{
		{"nil", func(context.Context, []model.Message) (*model.Response, error) { return nil, nil }},
		{"role", func(context.Context, []model.Message) (*model.Response, error) {
			return &model.Response{Message: model.Message{Role: "user"}}, nil
		}},
		{"unknown", func(context.Context, []model.Message) (*model.Response, error) { return request(), nil }},
	} {
		t.Run(tc.name, func(t *testing.T) {
			if _, e := must(t, tc.f, nil, 1).Run(context.Background(), "读取 doc-1"); e == nil {
				t.Fatal("expected error")
			}
		})
	}
	sentinel := errors.New("model failure")
	_, e := must(t, dModel(func(context.Context, []model.Message) (*model.Response, error) { return nil, sentinel }), nil, 1).Run(context.Background(), "")
	if !errors.Is(e, sentinel) {
		t.Fatal(e)
	}
	_, e = must(t, dModel(func(context.Context, []model.Message) (*model.Response, error) { return request(), nil }), map[string]tool.Tool{"get_document": dTool{func(context.Context, json.RawMessage) (string, error) { return "正文", nil }}}, 1).Run(context.Background(), "")
	if !errors.Is(e, agent.ErrMaxSteps) {
		t.Fatal(e)
	}
	_, e = must(t, dModel(func(context.Context, []model.Message) (*model.Response, error) { return request(), nil }), map[string]tool.Tool{"get_document": dTool{func(context.Context, json.RawMessage) (string, error) { return "", sentinel }}}, 2).Run(context.Background(), "")
	if !errors.Is(e, sentinel) {
		t.Fatal(e)
	}
	ctx, cancel := context.WithCancel(context.Background())
	cancel()
	_, e = must(t, m, nil, 1).Run(ctx, "")
	if !errors.Is(e, context.Canceled) {
		t.Fatal(e)
	}
}

package agent

import (
	"context"
	"errors"
	"fmt"
	"github.com/linko-zhao/mini-agent-runtime/internal/model"
	"github.com/linko-zhao/mini-agent-runtime/internal/tool"
)

var ErrMaxSteps = errors.New("agent: max steps exceeded")

type Agent struct {
	model    model.Model
	tools    map[string]tool.Tool
	maxSteps int
}

func New(m model.Model, tools map[string]tool.Tool, maxSteps int) (*Agent, error) {
	if m == nil || maxSteps <= 0 {
		return nil, fmt.Errorf("agent: model and positive maxSteps required")
	}
	a := &Agent{model: m, tools: make(map[string]tool.Tool), maxSteps: maxSteps}
	for n, t := range tools {
		if t == nil || n == "" || n != t.Name() {
			return nil, fmt.Errorf("agent: invalid registration")
		}
		a.tools[n] = t
	}
	return a, nil
}
func (a *Agent) Run(ctx context.Context, input string) (string, error) {
	messages := []model.Message{{Role: "user", Content: input}}
	for step := 1; step <= a.maxSteps; step++ {
		if err := ctx.Err(); err != nil {
			return "", err
		}
		r, err := a.model.Generate(ctx, messages)
		if err != nil {
			return "", err
		}
		if r == nil || r.Message.Role != "assistant" {
			return "", fmt.Errorf("agent: invalid model response")
		}
		messages = append(messages, r.Message)
		if len(r.Message.ToolCalls) == 0 {
			return r.Message.Content, nil
		}
		for _, call := range r.Message.ToolCalls {
			t, ok := a.tools[call.Name]
			if !ok {
				return "", fmt.Errorf("agent: unknown tool")
			}
			out, err := t.Execute(ctx, call.Arguments)
			if err != nil {
				return "", err
			}
			messages = append(messages, model.Message{Role: "tool", Content: out, ToolCallID: call.ID})
		}
	}
	return "", ErrMaxSteps
}

package model

import (
	"context"
	"fmt"
)

// Mock 是单个 Run 的脚本模型；并发 Run 应分别创建 Mock。
type Mock struct {
	responses []Response
	i         int
}

func NewMock(responses []Response) *Mock { return &Mock{responses: responses} }
func (m *Mock) Generate(ctx context.Context, _ []Message) (*Response, error) {
	if err := ctx.Err(); err != nil {
		return nil, err
	}
	if m.i >= len(m.responses) {
		return nil, fmt.Errorf("mock: exhausted responses")
	}
	r := m.responses[m.i]
	m.i++
	return &r, nil
}
