今天只解决什么？

实现 mcptool 适配器和 Discover，让远端能力实现 Day 08 的 Meta/Invoke，不重写 Loop。

两个服务器都可能提供 get_document；本地也有同名工具。远端返回内容块，而 Loop 只接收字符串。

在 Loop 中判断每种远端协议会不断增加分支。适配层负责名称和数据形状转换，Loop 继续按相同接口调用。

只投影文本块与对象 structuredContent，不假装支持图片。发现不修改活动注册表；不从 annotations 自动授予权限或启用安全重试。

本地别名与远端原名

模型看到 knowledge__get_document，远端请求仍使用 get_document。

前缀由可信宿主分配，区分不同服务器。

只有 ListTools 和 CallTool 两个方法。

这些字段、接口和方法为什么存在？

remoteTool 的三个字段

caller 保存调用端口，remoteName 用来路由原名，spec 保存带别名的描述快照。

Meta 只读快照，不请求远端。

New 的输入输出

New(caller,prefix,Descriptor) 返回 tool.Invokable。

构造拒绝 nil、非法名称、非 object schema；复制 InputSchema，不执行工具。

名称约束为何明确拒绝？

prefix 匹配 ^[A-Za-z][A-Za-z0-9-]*$，不允许下划线；原名匹配 ^[A-Za-z0-9_-]+$。

组合 prefix__name 最长 64 字节。

静默清洗可能让两个名字碰撞，所以不自动更名。

Discover 的页数和两个集合

maxPages 是正整数预算。

seenNames 检测跨页重名，seenCursors 检测游标循环；cursor 原样传回，不解析。

失败返回 nil，不泄露半套候选。

ResultEnvelope 的两个字段

Text 收集全部 text 文本，空 content 数组产生空 text 数组；StructuredContent 可缺省，存在时必须是对象并复制原 JSON。

只要有非文本块，整个投影失败，不保留部分文本伪装成功。

Invoke 的错误语义

取消/超时保留 context 原因并终止 Run。

RPC/传输错误和远端 IsError 返回 tool.ExecutionError 受控摘要，Cause 保留可信诊断；原错误正文不进入模型。

装配与执行分开

Initialize → Discover → 新 Registry → Tools → Policy → Agent。

Meta 不执行远端工具；本地授权名用别名，实际 tools/call 用原名；Principal 与模型调用 ID 不塞进 Arguments。

import (

type stubCaller struct {
	list func(context.Context, string) (mcp.ToolsPage, error)
	call func(context.Context, string, json.RawMessage) (mcp.ToolResult, error)
}

func (s stubCaller) ListTools(ctx context.Context, cursor string) (mcp.ToolsPage, error) {
	if s.list == nil {
		return mcp.ToolsPage{Tools: []mcp.Descriptor{}}, nil
	}
	return s.list(ctx, cursor)
}
func (s stubCaller) CallTool(ctx context.Context, name string, args json.RawMessage) (mcp.ToolResult, error) {
	if s.call == nil {
		return mcp.ToolResult{}, errors.New("unexpected execution")
	}
	return s.call(ctx, name, args)
}
func descriptor(name string) mcp.Descriptor {
	return mcp.Descriptor{Name: name, Description: "knowledge lookup", InputSchema: json.RawMessage(`{"type":"object"}`), Annotations: json.RawMessage(`{"readOnlyHint":true}`)}
}
func text(s string) json.RawMessage {
	b, _ := json.Marshal(map[string]string{"type": "text", "text": s})
	return b
}

func TestMetadataSnapshotAndNameBoundary(t *testing.T) {
	calls := 0
	caller := stubCaller{call: func(context.Context, string, json.RawMessage) (mcp.ToolResult, error) {
		calls++
		return mcp.ToolResult{}, nil
	}}
	d := descriptor("get_document")
	remote, err := New(caller, "knowledge", d)
	if err != nil {
		t.Fatal(err)
	}
	d.InputSchema[0] = 'x'
	spec, err := remote.Meta(context.Background())
	if err != nil || spec.Name != "knowledge__get_document" || !json.Valid(spec.InputSchema) || calls != 0 {
		t.Fatalf("%+v %v calls=%d", spec, err, calls)
	}
	spec.InputSchema[0] = 'x'
	again, _ := remote.Meta(context.Background())
	if !json.Valid(again.InputSchema) {
		t.Fatal("metadata mutation changed snapshot")
	}
	ctx, cancel := context.WithCancel(context.Background())
	cancel()
	if _, err := remote.Meta(ctx); !errors.Is(err, context.Canceled) {
		t.Fatal(err)
	}
	for _, tc := range []struct{ prefix, name string }{
		{"", "x"}, {"bad_prefix", "x"}, {"knowledge", "server.read"}, {"knowledge", "中文"},
		{"knowledge", strings.Repeat("x", 64)}, {"knowledge", ""},
	} {
		if _, err := New(caller, tc.prefix, descriptor(tc.name)); !errors.Is(err, ErrDiscovery) {
			t.Fatalf("%+v %v", tc, err)
		}
	}
	if _, err := New(nil, "knowledge", descriptor("x")); !errors.Is(err, ErrDiscovery) {
		t.Fatal(err)
	}
	bad := descriptor("x")
	bad.InputSchema = json.RawMessage(`{"type":"array"}`)
	if _, err := New(caller, "knowledge", bad); !errors.Is(err, ErrDiscovery) {
		t.Fatal(err)
	}
}

func TestDiscoveryPaginationAndAtomicFailure(t *testing.T) {
	var cursors []string
	caller := stubCaller{list: func(_ context.Context, cursor string) (mcp.ToolsPage, error) {
		cursors = append(cursors, cursor)
		if cursor == "" {
			return mcp.ToolsPage{Tools: []mcp.Descriptor{descriptor("search")}, NextCursor: "opaque-next"}, nil
		}
		return mcp.ToolsPage{Tools: []mcp.Descriptor{descriptor("read")}}, nil
	}}
	tools, err := Discover(context.Background(), caller, "knowledge", 2)
	if err != nil || len(tools) != 2 || strings.Join(cursors, ",") != ",opaque-next" {
		t.Fatalf("%v %v %v", tools, err, cursors)
	}
	first, _ := tools[0].Meta(context.Background())
	if first.Name != "knowledge__search" {
		t.Fatal(first)
	}
	tools, err = Discover(context.Background(), caller, "knowledge", 1)
	if tools != nil || !errors.Is(err, ErrDiscovery) {
		t.Fatalf("partial discovery: %v %v", tools, err)
	}
	loop := stubCaller{list: func(context.Context, string) (mcp.ToolsPage, error) {
		return mcp.ToolsPage{Tools: []mcp.Descriptor{}, NextCursor: "same"}, nil
	}}
	if tools, err = Discover(context.Background(), loop, "knowledge", 4); tools != nil || !errors.Is(err, ErrDiscovery) {
		t.Fatalf("%v %v", tools, err)
	}
	duplicate := stubCaller{list: func(_ context.Context, cursor string) (mcp.ToolsPage, error) {
		page := mcp.ToolsPage{Tools: []mcp.Descriptor{descriptor("read")}}
		if cursor == "" {
			page.NextCursor = "next"
		}
		return page, nil
	}}
	if tools, err = Discover(context.Background(), duplicate, "knowledge", 3); tools != nil || !errors.Is(err, ErrDiscovery) {
		t.Fatalf("%v %v", tools, err)
	}
	invalidPage := stubCaller{list: func(context.Context, string) (mcp.ToolsPage, error) { return mcp.ToolsPage{}, nil }}
	if _, err = Discover(context.Background(), invalidPage, "knowledge", 1); !errors.Is(err, ErrDiscovery) {
		t.Fatal(err)
	}
	ctx, cancel := context.WithCancel(context.Background())
	cancel()
	if _, err = Discover(ctx, caller, "knowledge", 2); !errors.Is(err, context.Canceled) {
		t.Fatal(err)
	}
}

func TestResultProjectionAndCancellation(t *testing.T) {
	caller := stubCaller{call: func(_ context.Context, name string, args json.RawMessage) (mcp.ToolResult, error) {
		if name != "get_document" || string(args) != `{"id":"doc-1"}` {
			t.Fatalf("%s %s", name, args)
		}
		args[0] = 'x'
		return mcp.ToolResult{Content: []json.RawMessage{text("first"), text("second")}, StructuredContent: json.RawMessage(`{"id":"doc-1"}`)}, nil
	}}
	remote, _ := New(caller, "knowledge", descriptor("get_document"))
	args := json.RawMessage(`{"id":"doc-1"}`)
	out, err := remote.Invoke(context.Background(), args)
	if err != nil || !json.Valid(args) {
		t.Fatalf("%q %v %s", out, err, args)
	}
	var result ResultEnvelope
	if json.Unmarshal([]byte(out), &result) != nil || strings.Join(result.Text, ",") != "first,second" || string(result.StructuredContent) != `{"id":"doc-1"}` {
		t.Fatal(out)
	}
	ctx, cancel := context.WithCancel(context.Background())
	late := stubCaller{call: func(context.Context, string, json.RawMessage) (mcp.ToolResult, error) {
		cancel()
		return mcp.ToolResult{Content: []json.RawMessage{text("late")}}, nil
	}}
	remote, _ = New(late, "knowledge", descriptor("read"))
	if out, err = remote.Invoke(ctx, nil); out != "" || !errors.Is(err, context.Canceled) {
		t.Fatalf("%q %v", out, err)
	}
}

func TestRemoteErrorsAreControlledAndNeverRetried(t *testing.T) {
	secret := "SECRET-backend-token"
	rpc := &mcp.RPCError{Code: -32603, Message: secret, Data: json.RawMessage(`{"private":"SECRET-backend-token"}`)}
	cases := []struct {
		name        string
		result      mcp.ToolResult
		cause, want error
	}{
		{"RPC", mcp.ToolResult{}, rpc, rpc},
		{"transport", mcp.ToolResult{}, errors.New(secret), nil},
		{"business", mcp.ToolResult{IsError: true, Content: []json.RawMessage{text(secret)}}, nil, ErrRemoteExecution},
		{"image", mcp.ToolResult{Content: []json.RawMessage{text("partial"), json.RawMessage(`{"type":"image","data":"AA=="}`)}}, nil, ErrUnsupportedContent},
		{"malformed text", mcp.ToolResult{Content: []json.RawMessage{json.RawMessage(`{"type":"text"}`)}}, nil, ErrInvalidResult},
		{"null structured", mcp.ToolResult{Content: []json.RawMessage{}, StructuredContent: json.RawMessage("null")}, nil, ErrInvalidResult},
		{"missing content", mcp.ToolResult{}, nil, ErrInvalidResult},
	}
	for _, tc := range cases {
		t.Run(tc.name, func(t *testing.T) {
			calls := 0
			caller := stubCaller{call: func(context.Context, string, json.RawMessage) (mcp.ToolResult, error) {
				calls++
				return tc.result, tc.cause
			}}
			remote, _ := New(caller, "knowledge", descriptor("read"))
			adapted, err := tool.Adapt(context.Background(), remote)
			if err != nil {
				t.Fatal(err)
			}
			wrapped, err := tool.WithPolicy(adapted, tool.Policy{MaxAttempts: 3, ReadOnly: true, Retryable: func(error) bool { return true }})
			if err != nil {
				t.Fatal(err)
			}
			out, err := wrapped.Execute(context.Background(), json.RawMessage("{}"))
			var known *tool.Error
			if out != "" || !errors.As(err, &known) || known.Kind != tool.ExecutionError || strings.Contains(err.Error(), secret) || calls != 1 {
				t.Fatalf("out=%q err=%v calls=%d", out, err, calls)
			}
			if tc.want != nil && !errors.Is(err, tc.want) {
				t.Fatal("lost diagnostic cause")
			}
			if tc.name == "RPC" {
				var nested *mcp.RPCError
				if !errors.As(err, &nested) || nested != rpc {
					t.Fatal("RPC diagnostic lost")
				}
			}
		})
	}
	for _, cause := range []error{context.Canceled, context.DeadlineExceeded} {
		caller := stubCaller{call: func(context.Context, string, json.RawMessage) (mcp.ToolResult, error) { return mcp.ToolResult{}, cause }}
		remote, _ := New(caller, "knowledge", descriptor("read"))
		if _, err := remote.Invoke(context.Background(), nil); !errors.Is(err, cause) {
			t.Fatal(err)
		}
	}
}

func TestDay13DiscoveryEdges(t *testing.T) {
	for _, tc := range []struct {
		caller Caller
		prefix string
		max    int
	}{{nil, "k", 1}, {stubCaller{}, "", 1}, {stubCaller{}, "k", 0}, {stubCaller{}, "k", -1}} {

	got, err := Discover(context.Background(), stubCaller{list: func(context.Context, string) (mcp.ToolsPage, error) { return mcp.ToolsPage{}, cause }}, "k", 1)
	if got != nil || !errors.Is(err, cause) {
		t.Fatalf("%+v %v", got, err)

// Package mcptool adapts the Day12 MCP client to the existing Day08 tool contract.
package mcptool

import (
	"context"
	"encoding/json"
	"errors"
	"fmt"
	"regexp"

	"github.com/linko-zhao/mini-agent-runtime/internal/mcp"
	"github.com/linko-zhao/mini-agent-runtime/internal/tool"
)

var (
	ErrDiscovery          = errors.New("mcptool: discovery rejected")
	ErrRemoteExecution    = errors.New("mcptool: remote execution failed")
	ErrUnsupportedContent = errors.New("mcptool: unsupported content")
	ErrInvalidResult      = errors.New("mcptool: invalid result")
	prefixPattern         = regexp.MustCompile(`^[A-Za-z][A-Za-z0-9-]*$`)
	namePattern           = regexp.MustCompile(`^[A-Za-z0-9_-]+$`)
)

// Caller is implemented by an already-initialized *mcp.Client.
// The host owns initialization, transport cleanup and recovery.
type Caller interface {
	ListTools(context.Context, string) (mcp.ToolsPage, error)
	CallTool(context.Context, string, json.RawMessage) (mcp.ToolResult, error)
}

// ResultEnvelope is the explicit projection into the existing string result.
// Non-text blocks are rejected, not silently flattened or discarded.
type ResultEnvelope struct {
	Text              []string        `json:"text"`
	StructuredContent json.RawMessage `json:"structuredContent,omitempty"`
}

type remoteTool struct {
	caller     Caller
	remoteName string
	spec       tool.Spec
}

func localName(prefix, name string) (string, error) {
	local := prefix + "__" + name
	// Prefixes deliberately exclude underscores so namespace boundaries are unique.
	if !prefixPattern.MatchString(prefix) || !namePattern.MatchString(name) || len(local) > 64 {
		return "", fmt.Errorf("%w: incompatible local tool name", ErrDiscovery)
	}
	return local, nil
}

// New snapshots metadata. It does not initialize a client, execute a tool,
// or derive authorization/retry policy from remote annotations.
func New(caller Caller, prefix string, d mcp.Descriptor) (tool.Invokable, error) {
	if caller == nil {
		return nil, fmt.Errorf("%w: nil caller", ErrDiscovery)
	}
	name, err := localName(prefix, d.Name)
	if err != nil {
		return nil, err
	}
	var schema map[string]any
	if json.Unmarshal(d.InputSchema, &schema) != nil || schema["type"] != "object" {
		return nil, fmt.Errorf("%w: object input schema required", ErrDiscovery)
	}
	return &remoteTool{caller: caller, remoteName: d.Name, spec: tool.Spec{
		Name: name, Description: d.Description, InputSchema: append(json.RawMessage(nil), d.InputSchema...),
	}}, nil
}
func (t *remoteTool) Meta(ctx context.Context) (tool.Spec, error) {
	if err := ctx.Err(); err != nil {
		return tool.Spec{}, err
	}
	spec := t.spec
	spec.InputSchema = append(json.RawMessage(nil), spec.InputSchema...)
	return spec, nil
}
func failure(message string, cause error) error {
	return &tool.Error{Kind: tool.ExecutionError, Message: message, Cause: cause}
}
func (t *remoteTool) Invoke(ctx context.Context, args json.RawMessage) (string, error) {
	if err := ctx.Err(); err != nil {
		return "", err
	}
	result, err := t.caller.CallTool(ctx, t.remoteName, append(json.RawMessage(nil), args...))
	if ctx.Err() != nil {
		return "", ctx.Err()
	}
	if errors.Is(err, context.Canceled) || errors.Is(err, context.DeadlineExceeded) {
		return "", err
	}
	if err != nil {
		return "", failure("remote tool request failed", err)
	}
	if result.IsError {
		return "", failure("remote tool execution failed", ErrRemoteExecution)
	}
	if result.Content == nil {
		return "", failure("invalid remote result", ErrInvalidResult)
	}
	envelope := ResultEnvelope{Text: make([]string, 0, len(result.Content))}
	for _, raw := range result.Content {
		var block struct {
			Type string  `json:"type"`
			Text *string `json:"text"`
		}
		if json.Unmarshal(raw, &block) != nil || block.Type == "" {
			return "", failure("invalid remote result", ErrInvalidResult)
		}
		if block.Type != "text" {
			return "", failure("remote content type is not supported", ErrUnsupportedContent)
		}
		if block.Text == nil {
			return "", failure("invalid remote result", ErrInvalidResult)
		}
		envelope.Text = append(envelope.Text, *block.Text)
	}
	if len(result.StructuredContent) > 0 {
		var object map[string]json.RawMessage
		if json.Unmarshal(result.StructuredContent, &object) != nil || object == nil {
			return "", failure("invalid remote result", ErrInvalidResult)
		}
		envelope.StructuredContent = append(json.RawMessage(nil), result.StructuredContent...)
	}
	output, err := json.Marshal(envelope)
	if err != nil {
		return "", failure("invalid remote result", ErrInvalidResult)
	}
	return string(output), nil
}

package mcptool

import (
	"context"
	"fmt"

	"github.com/linko-zhao/mini-agent-runtime/internal/tool"
)

// Discover returns all candidate adapters or nil on any failure. It never
// mutates a registry; the host assembles a new registry after discovery succeeds.
func Discover(ctx context.Context, caller Caller, prefix string, maxPages int) ([]tool.Invokable, error) {
	if caller == nil || !prefixPattern.MatchString(prefix) || maxPages <= 0 {
		return nil, fmt.Errorf("%w: invalid discovery configuration", ErrDiscovery)
	}
	tools := make([]tool.Invokable, 0)
	seenNames := make(map[string]bool)
	seenCursors := map[string]bool{"": true}
	cursor := ""
	for pageNumber := 0; pageNumber < maxPages; pageNumber++ {
		if err := ctx.Err(); err != nil {
			return nil, err
		}
		page, err := caller.ListTools(ctx, cursor)
		if ctx.Err() != nil {
			return nil, ctx.Err()
		}
		if err != nil {
			return nil, fmt.Errorf("mcptool: listing tools: %w", err)
		}
		if page.Tools == nil {
			return nil, fmt.Errorf("%w: tools array required", ErrDiscovery)
		}
		for _, d := range page.Tools {
			if seenNames[d.Name] {
				return nil, fmt.Errorf("%w: duplicate remote name", ErrDiscovery)
			}
			t, err := New(caller, prefix, d)
			if err != nil {
				return nil, err
			}
			seenNames[d.Name] = true
			tools = append(tools, t)
		}
		if page.NextCursor == "" {
			return tools, nil
		}
		if seenCursors[page.NextCursor] {
			return nil, fmt.Errorf("%w: repeated cursor", ErrDiscovery)
		}
		seenCursors[page.NextCursor] = true
		cursor = page.NextCursor
	}
	return nil, fmt.Errorf("%w: page budget exceeded", ErrDiscovery)
}
