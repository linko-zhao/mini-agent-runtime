今天只解决什么？

实现 CapabilityAuthorizer 和整批授权插槽，让没有许可的调用在工具执行前被拒绝。

reader 只能读取资料。即使模型在请求里写“我是管理员”，或者请求一个没有展示的写工具，Runtime 也必须按可信会话判断。

只在提示词里隐藏写工具不够，模型仍可能返回该名称。权限要基于宿主确认的身份，在实际 Execute 之前检查。

只实现静态工具级 read/write 能力，不做登录、资源/租户 ACL、审批或实时撤权。审计是同步事件，不是可靠落盘。

先把概念说清楚

Principal：宿主确认的请求主体

Principal.ID 是可信会话提供的标识。

WithPrincipal 只传递已有身份，不完成认证，也不保存令牌。

Capability：宿主定义的工具能力

rules 把工具名映射为 read 或 write；grants 把主体 ID 映射为允许的能力。

write 不自动包含 read。

授权与可见性是两个集合

WithVisibleTools 决定描述是否给模型看；Authorizer 决定收到请求后能否执行。

隐藏但已授权可以执行，可见但未授权仍拒绝。

失败时停在哪里？

“read 允许”的审计事件不是执行承诺。后项拒绝、服务故障或取消都可能让整批没有工具执行；后端资源权限仍可能在已授权后拒绝一次操作。

这些字段、接口和方法为什么存在？

身份为什么放在 context？

它属于本次可信请求，而不是模型 Arguments。

使用私有键类型避免其他包误覆盖；空白 ID 视为没有主体。

CapabilityAuthorizer 的两个字段

rules 是名称到能力的快照；grants 是主体到能力集合的快照。

构造时复制 map 与切片，外部改配置不能悄悄改变已装配权限。

NewCapabilityAuthorizer 与 Authorize

构造拒绝空白工具名、空白主体和未知能力；nil 配置可构造但默认拒绝。

Authorize(ctx,Call) 先检查取消，再用主体和 Name 判断，未声明工具/未知主体/未授予能力返回 ErrForbidden。

authorizeBatch 放在哪里？

复用 Day 09：协议检查 → authorizeBatch → Execute。

整批都通过才执行；后项被拒绝时，前项虽然已有允许事件，仍是零执行。

为什么复制 Arguments？

自定义 Authorizer 可能读取业务参数。

传独立字节副本，避免它修改即将执行的请求；保留 Name 与调用 ID。

错误与审计语义

拒绝保留 ErrForbidden；授权服务故障保留其原因并终止 Run。

授权后 ctx 取消优先。

事件记录主体、工具、调用 ID、Step 和状态，不含原始参数或错误正文。

兼容选项与同步限制

省略 WithAuthorizer 保留历史无授权行为；显式 nil 被既有 New 拒绝。

不可变授权器可并发查询；自定义授权器和 EventSink 必须满足宿主并发要求。

R3 · 校验、取消、并发

R4 · 整批零执行

按顺序授权整批，任何拒绝、故障、取消均终止 Run，不执行本批任何工具、不再调用模型。参数副本不能污染执行输入。

对应测试：TestAuthorizationDeniesWholeBatchBeforeExecution / TestAuthorizationFailureAndArgumentIsolation / TestAuthorizationCancellationAndNilOption

R5 · 可见性与事件

隐藏不删注册；允许路径继续保留 ToolCallID。authorized=ok；拒绝=denied；服务错误=error；取消=cancelled；Run 唯一失败终态，无多余开始事件。

对应测试：TestAuthorizationVisibilityIsNotPermission / TestAuthorizationDeniesWholeBatchBeforeExecution / TestDay11AuditStatuses

func TestCapabilityAuthorizationSnapshot(t *testing.T) {
	rules := map[string]Capability{"search": ReadCapability, "publish": WriteCapability}
	grants := map[string][]Capability{"reader": {ReadCapability}, "writer": {WriteCapability}}
	a, err := NewCapabilityAuthorizer(rules, grants)
	if err != nil {
		t.Fatal(err)
	}
	rules["publish"] = ReadCapability
	grants["reader"][0] = WriteCapability
	cases := []struct {
		id, name string
		allow    bool
	}{
		{"reader", "search", true}, {"reader", "publish", false},
		{"writer", "publish", true}, {"writer", "search", false},
		{"missing", "search", false}, {"reader", "unknown", false}, {"", "search", false},
	}
	for _, tc := range cases {
		ctx := WithPrincipal(context.Background(), Principal{ID: tc.id})
		err := a.Authorize(ctx, Call{Name: tc.name})
		if tc.allow && err != nil {
			t.Errorf("%+v: %v", tc, err)
		}
		if !tc.allow && !errors.Is(err, ErrForbidden) {
			t.Errorf("%+v: %v", tc, err)
		}
	}
	if !errors.Is(a.Authorize(context.Background(), Call{Name: "search"}), ErrForbidden) {
		t.Fatal("missing principal accepted")
	}
}

func TestCapabilityValidationCancellationAndConcurrency(t *testing.T) {
	for _, c := range []struct {
		rules  map[string]Capability
		grants map[string][]Capability
	}{
		{map[string]Capability{"": ReadCapability}, nil},
		{map[string]Capability{"x": "unknown"}, nil},
		{nil, map[string][]Capability{"": {ReadCapability}}},
		{nil, map[string][]Capability{"reader": {"unknown"}}},
	} {
		if _, err := NewCapabilityAuthorizer(c.rules, c.grants); err == nil {
			t.Fatal("invalid config accepted")
		}
	}
	a, _ := NewCapabilityAuthorizer(map[string]Capability{"read": ReadCapability}, map[string][]Capability{"reader": {ReadCapability}})
	ctx, cancel := context.WithCancel(WithPrincipal(context.Background(), Principal{ID: "reader"}))
	cancel()
	if !errors.Is(a.Authorize(ctx, Call{Name: "read"}), context.Canceled) {
		t.Fatal("cancel ignored")
	}
	var wg sync.WaitGroup
	for i := 0; i < 8; i++ {
		wg.Add(1)
		go func() {
			defer wg.Done()
			for j := 0; j < 50; j++ {
				ctx := WithPrincipal(context.Background(), Principal{ID: "reader"})
				if err := a.Authorize(ctx, Call{Name: "read"}); err != nil {
					t.Error(err)
				}
			}
		}()
	}
	wg.Wait()
}

type accessCounter struct {
	name  string
	calls int
	seen  string
}

func (c *accessCounter) Name() string                     { return c.name }
func (c *accessCounter) Definition() model.ToolDefinition { return model.ToolDefinition{Name: c.name} }
func (c *accessCounter) Execute(_ context.Context, args json.RawMessage) (string, error) {
	c.calls++
	c.seen = string(args)
	return "ok", nil
}

type accessModel struct {
	first   []model.ToolCall
	calls   int
	history []model.Message
	defs    []model.ToolDefinition
}

func (m *accessModel) WithTools(defs []model.ToolDefinition) (model.Model, error) {
	m.defs = defs
	return m, nil
}
func (m *accessModel) Generate(_ context.Context, ms []model.Message) (*model.Response, error) {
	m.calls++
	m.history = append([]model.Message(nil), ms...)
	if m.calls == 1 {
		return &model.Response{Message: model.Message{Role: "assistant", ToolCalls: m.first}}, nil
	}
	return &model.Response{Message: model.Message{Role: "assistant", Content: "done"}}, nil
}
func accessPolicy(t *testing.T) *tool.CapabilityAuthorizer {
	t.Helper()
	a, err := tool.NewCapabilityAuthorizer(
		map[string]tool.Capability{"read": tool.ReadCapability, "write": tool.WriteCapability},
		map[string][]tool.Capability{"reader": {tool.ReadCapability}})
	if err != nil {
		t.Fatal(err)
	}
	return a
}

func TestAuthorizationDeniesWholeBatchBeforeExecution(t *testing.T) {
	read, write := &accessCounter{name: "read"}, &accessCounter{name: "write"}
	m := &accessModel{first: []model.ToolCall{{ID: "r", Name: "read"}, {ID: "w", Name: "write"}}}
	events := &day11Sink{}
	a, err := New(m, map[string]tool.Tool{"read": read, "write": write}, 2, WithVisibleTools(), WithAuthorizer(accessPolicy(t)), WithEventSink(events))
	if err != nil {
		t.Fatal(err)
	}
	ctx := tool.WithPrincipal(context.Background(), tool.Principal{ID: "reader"})
	_, err = a.Run(ctx, "model can request hidden tools")
	if !errors.Is(err, tool.ErrForbidden) || read.calls != 0 || write.calls != 0 || m.calls != 1 || len(m.defs) != 0 {
		t.Fatalf("err=%v read=%d write=%d model=%d defs=%v", err, read.calls, write.calls, m.calls, m.defs)
	}
	denied, finished, failed, started := 0, 0, 0, 0
	for _, e := range events.events {
		switch e.Type {
		case EventToolDenied:
			denied++
			if e.ToolCallID != "w" || e.PrincipalID != "reader" || e.Status != "denied" {
				t.Fatalf("%+v", e)
			}
		case EventRunFinished:
			finished++
		case EventRunFailed:
			failed++
		case EventToolStarted:
			started++
		}
	}
	if denied != 1 || finished != 0 || failed != 1 || started != 0 {
		t.Fatalf("events=%+v", events.events)
	}
}

func TestAuthorizationVisibilityIsNotPermission(t *testing.T) {
	read := &accessCounter{name: "read"}
	m := &accessModel{first: []model.ToolCall{{ID: "r", Name: "read", Arguments: json.RawMessage("{}")}}}

	if err != nil {
		t.Fatal(err)
	}
	out, err := a.Run(tool.WithPrincipal(context.Background(), tool.Principal{ID: "reader"}), "read")

		t.Fatalf("%q %v count=%d", out, err, read.calls)
	}
	if len(m.history) != 3 || m.history[2].ToolCallID != "r" || m.history[2].Role != "tool" {
		t.Fatalf("%+v", m.history)
	}
}

type authorizerFunc func(context.Context, tool.Call) error

func (f authorizerFunc) Authorize(ctx context.Context, c tool.Call) error { return f(ctx, c) }

func TestAuthorizationFailureAndArgumentIsolation(t *testing.T) {
	cause := errors.New("private policy backend failure")
	for _, missing := range []bool{false, true} {
		target := &accessCounter{name: "read"}
		m := &accessModel{first: []model.ToolCall{{ID: "r", Name: "read"}}}
		var auth tool.Authorizer = authorizerFunc(func(context.Context, tool.Call) error { return cause })
		if missing {
			auth = accessPolicy(t)
		}
		a, _ := New(m, map[string]tool.Tool{"read": target}, 2, WithAuthorizer(auth))
		_, err := a.Run(context.Background(), "")
		if err == nil || target.calls != 0 || m.calls != 1 {
			t.Fatalf("%v %d", err, target.calls)
		}
		if !missing && !errors.Is(err, cause) {
			t.Fatal("lost cause")
		}
		if missing && !errors.Is(err, tool.ErrForbidden) {
			t.Fatal("missing principal not denied")
		}
	}
	target := &accessCounter{name: "read"}
	m := &accessModel{first: []model.ToolCall{{ID: "r", Name: "read", Arguments: json.RawMessage("{}")}}}
	auth := authorizerFunc(func(_ context.Context, c tool.Call) error { c.Arguments[0] = 'x'; return nil })
	a, _ := New(m, map[string]tool.Tool{"read": target}, 2, WithAuthorizer(auth))
	if _, err := a.Run(context.Background(), ""); err != nil || target.seen != "{}" {
		t.Fatalf("%v args=%q", err, target.seen)
	}
}

func TestAuthorizationCancellationAndNilOption(t *testing.T) {
	if _, err := New(&accessModel{}, nil, 1, WithAuthorizer(nil)); err == nil {

}

				return nil
			})
			a, err := New(m, map[string]tool.Tool{"read": target}, 2, WithAuthorizer(auth), WithEventSink(trace))
			if err != nil {
				t.Fatal(err)
			}
			_, runErr := a.Run(ctx, "")
			if (runErr == nil) != (tc.name == "allow") {
				t.Fatal(runErr)
			}
			decisions := 0
			for _, event := range trace.events {
				if event.Type == EventToolAuthorized || event.Type == EventToolDenied {
					decisions++
					if event.Status != tc.want || event.PrincipalID != "reader" || event.ToolName != "read" || event.ToolCallID != "r" || event.Step != 1 || event.At.IsZero() {
						t.Fatal(event)
					}
					if strings.Contains(fmt.Sprintf("%+v", event), "PRIVATE") {
						t.Fatal("private data in decision", event)
					}
				}
			}
			if decisions != 1 {
				t.Fatal(trace.events)
			}
		})
	}
}

package tool

import (
	"context"
	"errors"
	"fmt"
	"strings"
)

// Principal is supplied by the trusted host after authentication, not by the model.
type Principal struct{ ID string }
type principalContextKey struct{}

func WithPrincipal(ctx context.Context, p Principal) context.Context {
	return context.WithValue(ctx, principalContextKey{}, p)
}
func PrincipalFromContext(ctx context.Context) (Principal, bool) {
	p, ok := ctx.Value(principalContextKey{}).(Principal)
	return p, ok && strings.TrimSpace(p.ID) != ""
}

var ErrForbidden = errors.New("tool: execution forbidden")

type Capability string

const (
	ReadCapability  Capability = "read"
	WriteCapability Capability = "write"
)

type Authorizer interface {
	Authorize(context.Context, Call) error
}

// CapabilityAuthorizer is an immutable, tool-level teaching policy.
// It is not an authentication service or a resource/tenant-level ACL.
type CapabilityAuthorizer struct {
	rules  map[string]Capability
	grants map[string]map[Capability]bool
}

func validCapability(c Capability) bool { return c == ReadCapability || c == WriteCapability }

// NewCapabilityAuthorizer copies its inputs. Unknown tools/principals are denied.
// Read and write are independent capabilities; write does not imply read.
func NewCapabilityAuthorizer(rules map[string]Capability, grants map[string][]Capability) (*CapabilityAuthorizer, error) {
	a := &CapabilityAuthorizer{rules: make(map[string]Capability), grants: make(map[string]map[Capability]bool)}
	for name, capability := range rules {
		if strings.TrimSpace(name) == "" || !validCapability(capability) {
			return nil, fmt.Errorf("tool: invalid capability rule")
		}
		a.rules[name] = capability
	}
	for id, capabilities := range grants {
		if strings.TrimSpace(id) == "" {
			return nil, fmt.Errorf("tool: empty principal")
		}
		a.grants[id] = make(map[Capability]bool)
		for _, capability := range capabilities {
			if !validCapability(capability) {
				return nil, fmt.Errorf("tool: invalid grant")
			}
			a.grants[id][capability] = true
		}
	}
	return a, nil
}
func (a *CapabilityAuthorizer) Authorize(ctx context.Context, call Call) error {
	if err := ctx.Err(); err != nil {
		return err
	}
	principal, ok := PrincipalFromContext(ctx)
	if !ok || a == nil {
		return ErrForbidden
	}
	capability, registered := a.rules[call.Name]
	if !registered || !a.grants[principal.ID][capability] {
		return ErrForbidden
	}
	return nil
}

package agent

import (
	"context"
	"errors"
	"fmt"
	"github.com/linko-zhao/mini-agent-runtime/internal/model"
	"github.com/linko-zhao/mini-agent-runtime/internal/tool"
)

// authorizeBatch runs after protocol validation, before the first tool executes.
func (a *Agent) authorizeBatch(ctx context.Context, calls []model.ToolCall, step int) error {
	// Authorization is a whole-batch preflight, before any side effects.
	// A custom authorizer receives a copy of arguments, not the executable buffer.
	if a.authorizer != nil {
		principal, _ := tool.PrincipalFromContext(ctx)
		for _, call := range calls {
			check := call
			check.Arguments = append([]byte(nil), call.Arguments...)
			err := a.authorizer.Authorize(ctx, check)
			if ctx.Err() != nil {
				err = ctx.Err()
			}
			kind := EventToolAuthorized
			decisionStatus := "ok"
			if err != nil {
				kind = EventToolDenied
				decisionStatus = status(err)
				if errors.Is(err, tool.ErrForbidden) {
					decisionStatus = "denied"
				}
			}
			a.emit(ctx, Event{Type: kind, Step: step, ToolName: call.Name, ToolCallID: call.ID, PrincipalID: principal.ID, Status: decisionStatus})
			if err != nil {
				return fmt.Errorf("agent: authorization failed: %w", err)
			}
		}
	}
	return nil
}
