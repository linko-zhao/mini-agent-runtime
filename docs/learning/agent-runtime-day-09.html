今天只解决什么？

实现 Agent.executeBatch，让模型一次提出的多个工具调用，得到可预测的执行顺序和结果关联。

用户已经知道文档 a、b 的 ID，希望同时读取两份正文。模型返回两条 ToolCall，第一条合法，第二条却可能写错名称。

如果检查一条就执行一条，发现第二条有错时，第一条已经执行。把“整批检查”与“真正执行”分开，才能在协议错误时做到本批零执行。

今天实现批次边界，不解析业务参数，不增加并行、重试或事务。复用前序事件、超时和消息；为 Day 11 保留预检查插槽。

先把概念说清楚

批次：模型一次返回的调用列表

同一条 assistant 消息可以携带多个 ToolCall。

本例是 read-1 读取 a，read-2 读取 b；不是两次模型请求。

预检查：先确认整批能被识别

检查每个 ID 非空且批内唯一、每个工具名都已注册。

只检查名称和 ID；文档 id 是否有效由工具判断。

结果关联：按调用 ID 回填

结果消息 Role=tool，ToolCallID 等于原请求 ID。

工具名可以相同，调用 ID 必须区分这两次读取。

失败时停在哪里？

第二条请求名称为 missing 时，整批报错，第一条也不执行。若第一条已经实际执行、第二条发生运行时取消，第一条的副作用不会因此撤销。

这些字段、接口和方法为什么存在？

为什么方法属于 Agent？

executeBatch(ctx, calls, step) 使用 Agent 的工具表、超时和事件配置，返回 []model.Message 或终止错误。

它不增加第二套 Call/Result 协议。

calls 与 step 各是什么？

calls 是当前 assistant 的 ToolCalls，按输入顺序执行；step 是模型轮次，从 1 开始，只用于事件。

不要把内部尝试计入 step。

为什么分两个循环？

第一个循环检查整批 ID/名称；成功后，第二个循环才执行。

结构错误返回 nil,error，不把已经生成的一部分结果交回模型。

预留 authorizeBatch 插槽

协议检查成功后、首个 Execute 前，调用提供的 authorizeBatch(ctx,calls,step)。

它无配置时直接放行；今天复用，Day 11 再实现可信授权。

为什么要单独检查 context？

每个工具执行前检查父 ctx；有 toolTimeout 时派生工具 ctx，执行后检查它是否已取消，再释放计时器。

工具即使误报成功，取消仍优先。

为什么错误不能全按一种方式处理？

参数/业务错误回填安全提示；取消和超时终止 Run。

未知内部错误使用固定摘要，不把后端令牌或堆栈送进模型。

事件怎样配对？

每个已开始的调用发布 started 和 completed，包含 Step、ToolName、ToolCallID、耗时与状态。

预检查失败不发布 tool started；Run 的唯一终态由既有 Run 方法管理。

package agent

import (
	"context"
	"errors"
	"fmt"
	"github.com/linko-zhao/mini-agent-runtime/internal/model"
	"github.com/linko-zhao/mini-agent-runtime/internal/tool"
)

// authorizeBatch runs after protocol validation, before the first tool executes.
func (a *Agent) authorizeBatch(ctx context.Context, calls []model.ToolCall, step int) error {
	// Authorization is a whole-batch preflight, before any side effects.
	// A custom authorizer receives a copy of arguments, not the executable buffer.
	if a.authorizer != nil {
		principal, _ := tool.PrincipalFromContext(ctx)
		for _, call := range calls {
			check := call
			check.Arguments = append([]byte(nil), call.Arguments...)
			err := a.authorizer.Authorize(ctx, check)
			if ctx.Err() != nil {
				err = ctx.Err()
			}
			kind := EventToolAuthorized
			decisionStatus := "ok"
			if err != nil {
				kind = EventToolDenied
				decisionStatus = status(err)
				if errors.Is(err, tool.ErrForbidden) {
					decisionStatus = "denied"
				}
			}
			a.emit(ctx, Event{Type: kind, Step: step, ToolName: call.Name, ToolCallID: call.ID, PrincipalID: principal.ID, Status: decisionStatus})
			if err != nil {
				return fmt.Errorf("agent: authorization failed: %w", err)
			}
		}
	}
	return nil
}

func TestDay09BatchRoundTrip(t *testing.T) {
	calls := day09Calls()
	rounds := 0
	var seen []string
	sink := &day09Sink{}
	m := day09Model(func(_ context.Context, ms []model.Message) (*model.Response, error) {
		rounds++
		if rounds == 1 {
			return &model.Response{Message: model.Message{Role: "assistant", ToolCalls: calls}}, nil
		}
		if len(ms) != 4 || ms[1].Role != "assistant" || len(ms[1].ToolCalls) != 2 {
			t.Fatalf("missing assistant: %+v", ms)
		}
		for i, id := range []string{"read-1", "read-2"} {
			if ms[i+2].Role != "tool" || ms[i+2].ToolCallID != id || ms[i+2].Content != []string{"a", "b"}[i] {
				t.Fatalf("wrong pairing: %+v", ms)
			}
		}
		return &model.Response{Message: model.Message{Role: "assistant", Content: "done"}}, nil
	})
	target := day09Tool{"read", func(_ context.Context, args json.RawMessage) (string, error) {
		var p struct {
			ID string `json:"id"`
		}
		if err := json.Unmarshal(args, &p); err != nil {
			t.Fatal(err)
		}
		seen = append(seen, p.ID)
		return p.ID, nil
	}}
	a := day09New(t, m, map[string]tool.Tool{"read": target}, agent.WithEventSink(sink))
	out, err := a.Run(context.Background(), "read a and b")
	if err != nil || out != "done" || rounds != 2 || !reflect.DeepEqual(seen, []string{"a", "b"}) {
		t.Fatalf("%q %v seen=%v rounds=%d", out, err, seen, rounds)
	}
	started, completed, terminal := 0, 0, 0
	for _, e := range sink.events {
		switch e.Type {
		case agent.EventToolStarted:
			started++
		case agent.EventToolCompleted:
			completed++
			if e.Status != "ok" || e.Step != 1 || e.At.IsZero() {
				t.Fatal(e)
			}
		case agent.EventRunFinished, agent.EventRunFailed:
			terminal++
		}
	}
	if started != 2 || completed != 2 || terminal != 1 {
		t.Fatalf("events=%+v", sink.events)
	}
}

func TestDay09CancellationAndTimeout(t *testing.T) {
	for _, deadline := range []bool{false, true} {
		t.Run(map[bool]string{false: "cancel", true: "timeout"}[deadline], func(t *testing.T) {
			ctx, cancel := context.WithCancel(context.Background())
			defer cancel()
			n, rounds := 0, 0
			sink := &day09Sink{}
			m := day09Model(func(context.Context, []model.Message) (*model.Response, error) {
				rounds++
				return &model.Response{Message: model.Message{Role: "assistant", ToolCalls: day09Calls()}}, nil
			})
			target := day09Tool{"read", func(c context.Context, _ json.RawMessage) (string, error) {
				n++
				if deadline {
					<-c.Done()
				} else {
					cancel()
				}
				return "late success", nil
			}}
			opts := []agent.Option{agent.WithEventSink(sink)}
			if deadline {
				opts = append(opts, agent.WithToolTimeout(time.Millisecond))
			}
			a := day09New(t, m, map[string]tool.Tool{"read": target}, opts...)
			_, err := a.Run(ctx, "")
			want := context.Canceled
			if deadline {
				want = context.DeadlineExceeded
			}
			if !errors.Is(err, want) || n != 1 || rounds != 1 {
				t.Fatalf("%v calls=%d rounds=%d", err, n, rounds)
			}
			completed := 0
			for _, e := range sink.events {
				if e.Type == agent.EventToolCompleted {
					completed++
					if e.Status != "cancelled" {
						t.Fatal(e)
					}
				}
			}
			if completed != 1 {
				t.Fatal(sink.events)
			}
		})
	}
}

package agent

import (
	"context"
	"errors"
	"fmt"
	"time"

	"github.com/linko-zhao/mini-agent-runtime/internal/model"
	"github.com/linko-zhao/mini-agent-runtime/internal/tool"
)

// executeBatch validates the entire batch before executing any tool.
// Business arguments remain the tool's responsibility. Results preserve input
// order and call IDs; cancellation or timeout terminates the batch with an error.
func (a *Agent) executeBatch(ctx context.Context, calls []model.ToolCall, step int) ([]model.Message, error) {
	var results []model.Message
	// 整批预校验，避免后面的未知工具或重复 ID 导致前面的工具已产生副作用。
	seen := map[string]bool{}
	for _, call := range calls {
		if call.ID == "" || seen[call.ID] {
			return nil, fmt.Errorf("agent: missing or duplicate tool call ID")
		}
		seen[call.ID] = true
		if _, ok := a.tools[call.Name]; !ok {
			return nil, &tool.Error{Kind: tool.NotFoundError, Message: "tool not registered"}
		}
	}
	if err := a.authorizeBatch(ctx, calls, step); err != nil {
		return nil, err
	}
	for _, call := range calls {
		if err := ctx.Err(); err != nil {
			return nil, err
		}
		start := time.Now()
		e := Event{Type: EventToolStarted, Step: step, ToolName: call.Name, ToolCallID: call.ID, Status: "ok"}
		a.emit(ctx, e)
		toolCtx := ctx
		cancel := func() {}
		if a.toolTimeout > 0 {
			toolCtx, cancel = context.WithTimeout(ctx, a.toolTimeout)
		}
		result, err := a.tools[call.Name].Execute(toolCtx, call.Arguments)
		if toolCtx.Err() != nil {
			err = toolCtx.Err()
		}
		cancel()
		e.Type = EventToolCompleted
		e.Duration = time.Since(start)
		e.Status = status(err)
		a.emit(ctx, e)
		if errors.Is(err, context.Canceled) || errors.Is(err, context.DeadlineExceeded) {
			return nil, err
		}
		if err != nil {
			// 仅回填受控提示，未知内部错误不暴露给模型。
			var known *tool.Error
			if errors.As(err, &known) {
				result = known.Error()
			} else {
				result = "execution_failed: tool execution failed"
			}
		}
		results = append(results, model.Message{Role: "tool", ToolCallID: call.ID, Content: result})
	}
	return results, nil
}
