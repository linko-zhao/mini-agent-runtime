今天只解决什么

今天实现合作式取消和单工具超时，让取消后的读取流程停止继续推进。

工具可能在取消后误报成功，模型也可能返回一个迟到的回答。只把 context 传下去，不足以保证 Loop 不继续执行后续动作。

复用 Day03 工具绑定；不安排补课门禁。只控制串行边界，不强杀 goroutine、不回滚外部操作、不重试。

先解释概念，再看一次具体运行

合作式取消

是什么：传递停止信号，由当前操作主动响应。

为什么需要：get_document 等待 ctx.Done 后返回。

落到代码：Run 前后检查 ctx，依赖收到同一个或派生 context。

单工具超时

是什么：每次执行工具时独立计算的一段时间预算。

为什么需要：读取 doc-1 超时后，不再调用下一轮模型。

落到代码：WithToolTimeout 配置 time.Duration；WithTimeout 派生执行 context。

模型轮次预算

是什么：限制 Generate 最多调用几次。

为什么需要：maxSteps=1 时读取完成也不能再问模型。

落到代码：沿用 ErrMaxSteps，不把工具个数算作轮次。

失败反例

若工具已经发出写请求，取消只能停止后续推进，不能证明写入没有发生。非合作工具仍可能阻塞；本课没有隔离或强制终止承诺。

为什么这样设计

Option / WithToolTimeout

职责与字段/参数：New 增加 ...Option，旧三参数调用仍合法。Option 是修改构造中 Agent 配置的函数；nil Option 忽略。WithToolTimeout(d) 返回选项，d 单位纳秒的 time.Duration。

返回与失败：d=0 禁用单工具超时；d<0 由 New 返回 nil,error；正值应用到每次 Execute。

状态、所有权与取消：选项仅构造时应用，运行时不得修改配置；工具对象仍共享。

设计理由：在入口统一配置生命周期，避免每个业务工具重复算预算。

Agent.toolTimeout / Run

职责与字段/参数：toolTimeout 保存配置；每个 Run 的 messages/step 保持私有。模型前检查父 ctx，模型返回后父取消优先于成功或其他错误。

返回与失败：工具前检查父 ctx，传入派生 ctx；返回后优先检查 toolCtx.Err，再释放 cancel。取消/超时 errors.Is 保留，不把 late 当正文。

状态、所有权与取消：父 context 同样生效；零超时直接传父 ctx。可能已有副作用，不回滚；无并发承诺。

设计理由：双边检查防止不规范依赖误报成功。

New 与文件迁移

职责与字段/参数：New 仍负责模型、工具和步数校验，并接受选项；Run 移到 run.go，Agent 定义保留在 agent.go。

返回与失败：页面提供完整支持文件 new.go，主要新增行为只是应用选项与拒绝负超时。

状态、所有权与取消：手工迁移从旧 agent.go 删除 Run 后才新增 run.go；保持 Day03 的 New 只在 new.go 定义一次。

设计理由：拆分文件让本日骨架聚焦执行边界。

行为契约：不用读答案也知道怎么写

按编号实现；未规定的错误文本不需逐字匹配。依赖错误用 errors.Is/As 判断，不以私有字段或某种容器为验收目标。

R1 · 取消优先

已取消不调用模型；模型中取消但返回成功仍取消；第一个工具中取消后第二个工具和下一轮模型都不执行，零超时保留原 ctx。

package agent

import (
	"fmt"
	"github.com/linko-zhao/mini-agent-runtime/internal/model"
	"github.com/linko-zhao/mini-agent-runtime/internal/tool"
	"sort"
)

func New(m model.Model, tools map[string]tool.Tool, maxSteps int, opts ...Option) (*Agent, error) {
	if m == nil || maxSteps <= 0 {
		return nil, fmt.Errorf("agent: model and positive maxSteps required")
	}
	a := &Agent{model: m, tools: make(map[string]tool.Tool), maxSteps: maxSteps}
	for _, o := range opts {
		if o != nil {
			o(a)
		}
	}
	if a.toolTimeout < 0 {
		return nil, fmt.Errorf("agent: negative timeout")
	}
	names := []string{}
	for n, t := range tools {
		if t == nil || n == "" || n != t.Name() {
			return nil, fmt.Errorf("agent: invalid registration")
		}
		a.tools[n] = t
		names = append(names, n)
	}
	sort.Strings(names)
	var defs []model.ToolDefinition
	for _, n := range names {
		if t, ok := a.tools[n].(tool.DescribedTool); ok {
			d := t.Definition()
			if d.Name != n {
				return nil, fmt.Errorf("agent: definition name mismatch")
			}
			defs = append(defs, d)
		}
	}
	if b, ok := m.(model.ToolBinder); ok {
		bound, err := b.WithTools(defs)
		if err != nil {
			return nil, err
		}
		if bound == nil {
			return nil, fmt.Errorf("agent: nil bound model")
		}
		a.model = bound
	}
	return a, nil
}

package agent_test

import (
	"context"
	"encoding/json"
	"errors"
	"github.com/linko-zhao/mini-agent-runtime/internal/agent"
	"github.com/linko-zhao/mini-agent-runtime/internal/model"
	"github.com/linko-zhao/mini-agent-runtime/internal/tool"
	"testing"
	"time"
)

type day04Model func(context.Context, []model.Message) (*model.Response, error)

func (f day04Model) Generate(c context.Context, m []model.Message) (*model.Response, error) {
	return f(c, m)
}

type day04Tool struct {
	f func(context.Context, json.RawMessage) (string, error)
}

func (day04Tool) Name() string                                                   { return "get_document" }
func (f day04Tool) Execute(c context.Context, a json.RawMessage) (string, error) { return f.f(c, a) }
func day04Answer(s string) *model.Response {
	return &model.Response{Message: model.Message{Role: "assistant", Content: s}}
}
func day04Request() *model.Response {
	return &model.Response{Message: model.Message{Role: "assistant", ToolCalls: []model.ToolCall{{ID: "read-1", Name: "get_document", Arguments: json.RawMessage(`{"id":"doc-1"}`)}}}}
}
func day04Must(t *testing.T, m model.Model, tools map[string]tool.Tool, n int) *agent.Agent {
	t.Helper()
	a, e := agent.New(m, tools, n)
	if e != nil {
		t.Fatal(e)
	}
	return a
}

func TestDay04Cancellation(t *testing.T) {
	calls := 0
	m := day04Model(func(context.Context, []model.Message) (*model.Response, error) {
		calls++
		return day04Answer("late"), nil
	})
	a, e := agent.New(m, nil, 1)
	if e != nil {
		t.Fatal(e)
	}
	ctx, cancel := context.WithCancel(context.Background())
	cancel()
	_, e = a.Run(ctx, "")
	if !errors.Is(e, context.Canceled) || calls != 0 {
		t.Fatal(e, calls)
	}
	ctx, cancel = context.WithCancel(context.Background())
	m = day04Model(func(context.Context, []model.Message) (*model.Response, error) {
		cancel()
		return day04Answer("late"), nil
	})
	a, _ = agent.New(m, nil, 1)
	if _, e = a.Run(ctx, ""); !errors.Is(e, context.Canceled) {
		t.Fatal(e)
	}
	ctx, cancel = context.WithCancel(context.Background())
	reads := 0
	rounds := 0
	m = day04Model(func(context.Context, []model.Message) (*model.Response, error) {
		rounds++
		r := day04Request()
		r.Message.ToolCalls = append(r.Message.ToolCalls, model.ToolCall{ID: "read-2", Name: "get_document", Arguments: json.RawMessage(`{}`)})
		return r, nil
	})
	a, e = agent.New(m, map[string]tool.Tool{"get_document": day04Tool{func(c context.Context, _ json.RawMessage) (string, error) {
		reads++
		if c != ctx {
			t.Fatal("parent ctx lost")
		}
		cancel()
		return "late", nil
	}}}, 2)
	if e != nil {
		t.Fatal(e)
	}
	_, e = a.Run(ctx, "")
	if !errors.Is(e, context.Canceled) || reads != 1 || rounds != 1 {
		t.Fatal(e, reads, rounds)
	}
}
func TestDay04Timeout(t *testing.T) {
	m := day04Model(func(context.Context, []model.Message) (*model.Response, error) { return day04Request(), nil })
	if _, e := agent.New(m, nil, 1, agent.WithToolTimeout(-1)); e == nil {
		t.Fatal("negative timeout")
	}
	reads := 0
	a, e := agent.New(m, map[string]tool.Tool{"get_document": day04Tool{func(c context.Context, _ json.RawMessage) (string, error) {
		reads++
		if _, ok := c.Deadline(); !ok {
			t.Fatal("missing deadline")
		}
		<-c.Done()
		return "late", nil
	}}}, 2, agent.WithToolTimeout(time.Nanosecond))
	if e != nil {
		t.Fatal(e)
	}
	_, e = a.Run(context.Background(), "")
	if !errors.Is(e, context.DeadlineExceeded) || reads != 1 {
		t.Fatal(e, reads)
	}
	a, e = agent.New(day04Model(func(context.Context, []model.Message) (*model.Response, error) { return day04Answer("ok"), nil }), nil, 1, agent.WithToolTimeout(0))
	if e != nil {
		t.Fatal(e)
	}
	for i := 0; i < 2; i++ {
		if s, e := a.Run(context.Background(), ""); e != nil || s != "ok" {
			t.Fatal(s, e)
		}
	}
}

package agent

import (
	"context"
	"fmt"
	"github.com/linko-zhao/mini-agent-runtime/internal/model"
)

func (a *Agent) Run(ctx context.Context, input string) (string, error) {
	messages := []model.Message{{Role: "user", Content: input}}
	for step := 1; step <= a.maxSteps; step++ {
		if e := ctx.Err(); e != nil {
			return "", e
		}
		r, e := a.model.Generate(ctx, messages)
		if ctx.Err() != nil {
			e = ctx.Err()
		}
		if e != nil {
			return "", e
		}
		if r == nil || r.Message.Role != "assistant" {
			return "", fmt.Errorf("agent: invalid response")
		}
		messages = append(messages, r.Message)
		if len(r.Message.ToolCalls) == 0 {
			return r.Message.Content, nil
		}
		for _, call := range r.Message.ToolCalls {
			if e := ctx.Err(); e != nil {
				return "", e
			}
			t, ok := a.tools[call.Name]
			if !ok {
				return "", fmt.Errorf("agent: unknown tool")
			}
			tc := ctx
			cancel := func() {}
			if a.toolTimeout > 0 {
				tc, cancel = context.WithTimeout(ctx, a.toolTimeout)
			}
			out, e := t.Execute(tc, call.Arguments)
			if tc.Err() != nil {
				e = tc.Err()
			}
			cancel()
			if e != nil {
				return "", e
			}
			messages = append(messages, model.Message{Role: "tool", Content: out, ToolCallID: call.ID})
		}
	}
	return "", ErrMaxSteps
}

type Option func(*Agent)
