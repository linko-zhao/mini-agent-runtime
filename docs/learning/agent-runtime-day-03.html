今天只解决什么

今天实现工具说明绑定，让模型在执行前知道 get_document 能做什么，同时保留独立的执行对象。

Loop 已能执行 get_document，但真实模型若没有收到工具说明，就无法可靠地产生该名称和参数。绑定若原地修改模型，两个宿主还可能互相污染工具集合。

实现非流式工具说明和一次装配绑定；不连真实模型，不实现权限、流式工具、注册表或执行策略。

先解释概念，再看一次具体运行

工具说明

是什么：模型选择能力时读取的数据。

为什么需要：get_document 的名称、描述和 object schema。

落到代码：ToolDefinition 保存 Name/Description/Parameters。

可选绑定

是什么：只有支持工具协议的模型才接受说明。

为什么需要：脚本模型仍可只实现 Generate。

落到代码：ToolBinder.WithTools 返回另一个 Model。

装配

是什么：运行之前连接依赖的过程。

为什么需要：宿主收集说明后只绑定一次。

落到代码：Agent.New 排序、校验并保存绑定后的模型。

失败反例

两个说明重名，或 Parameters 不是 type=object 的 JSON 对象，WithTools 返回 nil,error。绑定失败时 New 返回 nil,error，不进入 Run。

为什么这样设计

ToolDefinition / DescribedTool

职责与字段/参数：Name 为稳定工具名；Description 可空；Parameters 是自有 JSON schema 字节，要求顶层对象且 type=object。DescribedTool 嵌入 Tool 并增加 Definition()。

返回与失败：Definition 返回说明；它不读取业务正文。未提供说明的旧 Tool 仍可执行，但不出现在说明列表。

状态、所有权与取消：宿主不修改工具本身；调用方不要在装配期间并发修改 Definition 来源。

设计理由：说明是数据，执行对象含行为，两者不能混为权限。

ToolBinder.WithTools

职责与字段/参数：WithTools(defs) 接收按调用方顺序排列的说明，nil/空清空绑定。返回新 Model，不修改原 Adapter。

返回与失败：空名、重名、无效/非 object schema 返回 nil,error；不返回部分绑定；不强制 Description 或额外的 JSON Schema 语义。

状态、所有权与取消：深拷贝 Parameters，复制切片和值；构造后调用方改原字段/字节不影响新模型。无网络，无并发承诺。

设计理由：避免多个 Agent 共享模型时互相覆盖说明。

Agent.New

职责与字段/参数：New 的输入仍是 m、工具 map、正步数。按名称排序可描述工具，Definition.Name 必须等于注册名；支持 ToolBinder 时仅绑定一次。

返回与失败：保存 WithTools 返回的模型；错误保留原因链；返回 nil 模型也拒绝。普通 Model 不强制实现 ToolBinder。

状态、所有权与取消：沿用复制 map、不可变装配；Run 复用绑定产物。typed nil 不在验证范围。

设计理由：装配失败早于任何业务执行，排序使请求稳定。

package agent

import (
	"context"
	"errors"
	"fmt"
	"github.com/linko-zhao/mini-agent-runtime/internal/model"
	"github.com/linko-zhao/mini-agent-runtime/internal/tool"
)

var ErrMaxSteps = errors.New("agent: max steps exceeded")

type Agent struct {
	model    model.Model
	tools    map[string]tool.Tool
	maxSteps int
}

func (a *Agent) Run(ctx context.Context, input string) (string, error) {
	messages := []model.Message{{Role: "user", Content: input}}
	for step := 1; step <= a.maxSteps; step++ {
		if err := ctx.Err(); err != nil {
			return "", err
		}
		r, err := a.model.Generate(ctx, messages)
		if err != nil {
			return "", err
		}
		if r == nil || r.Message.Role != "assistant" {
			return "", fmt.Errorf("agent: invalid model response")
		}
		messages = append(messages, r.Message)
		if len(r.Message.ToolCalls) == 0 {
			return r.Message.Content, nil
		}
		for _, call := range r.Message.ToolCalls {
			t, ok := a.tools[call.Name]
			if !ok {
				return "", fmt.Errorf("agent: unknown tool")
			}
			out, err := t.Execute(ctx, call.Arguments)
			if err != nil {
				return "", err
			}
			messages = append(messages, model.Message{Role: "tool", Content: out, ToolCallID: call.ID})
		}
	}
	return "", ErrMaxSteps
}

package openai_test

import (
	"context"

	"github.com/linko-zhao/mini-agent-runtime/internal/model"

type day03Transport func(*http.Request) (*http.Response, error)

func (f day03Transport) RoundTrip(r *http.Request) (*http.Response, error) { return f(r) }
func day03Adapter(t *testing.T, f day03Transport) *openai.Adapter {
	t.Helper()
	a, e := openai.New(openai.Config{APIKey: "test", Model: "course", BaseURL: "https://provider.invalid/chat", Client: &http.Client{Transport: f}})
	if e != nil {
		t.Fatal(e)
	}
	return a
}
func day03Response(status int, s string) *http.Response {
	return &http.Response{StatusCode: status, Body: io.NopCloser(strings.NewReader(s)), Header: make(http.Header)}
}

func TestDay03Binding(t *testing.T) {
	var bodies []map[string]any
	a := day03Adapter(t, func(r *http.Request) (*http.Response, error) {
		var b map[string]any
		if e := json.NewDecoder(r.Body).Decode(&b); e != nil {
			t.Fatal(e)
		}
		bodies = append(bodies, b)
		return day03Response(200, `{"choices":[{"message":{"role":"assistant","content":"已读取"}}]}`), nil
	})
	defs := []model.ToolDefinition{{Name: "get_document", Description: "读取文档", Parameters: json.RawMessage(`{"type":"object"}`)}}
	bound, e := a.WithTools(defs)
	if e != nil {
		t.Fatal(e)
	}
	defs[0].Name = "changed"
	defs[0].Parameters[0] = 'x'
	if _, e = a.Generate(context.Background(), nil); e != nil {
		t.Fatal(e)
	}
	if _, e = bound.Generate(context.Background(), nil); e != nil {
		t.Fatal(e)
	}
	if _, ok := bodies[0]["tools"]; ok {
		t.Fatal("original mutated")
	}
	ts := bodies[1]["tools"].([]any)
	f := ts[0].(map[string]any)["function"].(map[string]any)
	if len(ts) != 1 || ts[0].(map[string]any)["type"] != "function" || f["name"] != "get_document" || f["description"] != "读取文档" || f["parameters"].(map[string]any)["type"] != "object" {
		t.Fatal(ts)
	}
	for _, ds := range [][]model.ToolDefinition{{{Name: "", Parameters: json.RawMessage(`{}`)}}, {{Name: "a", Parameters: json.RawMessage(`[]`)}}, {{Name: "a", Parameters: json.RawMessage(`{"type":"array"}`)}}, {{Name: "a", Parameters: json.RawMessage(`bad`)}}, {{Name: "a", Parameters: json.RawMessage(`{"type":"object"}`)}, {Name: "a", Parameters: json.RawMessage(`{"type":"object"}`)}}} {
		if _, e := a.WithTools(ds); e == nil {
			t.Fatal("invalid definition")
		}
	}
	if b, e := a.WithTools(nil); e != nil || b == nil {
		t.Fatal(b, e)
	}
}

package agent_test

import (
	"context"
	"encoding/json"
	"errors"
	"github.com/linko-zhao/mini-agent-runtime/internal/agent"
	"github.com/linko-zhao/mini-agent-runtime/internal/model"
	"github.com/linko-zhao/mini-agent-runtime/internal/tool"
	"testing"
)

type day03Model func(context.Context, []model.Message) (*model.Response, error)

func (f day03Model) Generate(c context.Context, m []model.Message) (*model.Response, error) {
	return f(c, m)
}

type day03Tool struct {
	f func(context.Context, json.RawMessage) (string, error)
}

func (day03Tool) Name() string                                                   { return "get_document" }
func (f day03Tool) Execute(c context.Context, a json.RawMessage) (string, error) { return f.f(c, a) }
func day03Answer(s string) *model.Response {
	return &model.Response{Message: model.Message{Role: "assistant", Content: s}}
}
func day03Request() *model.Response {
	return &model.Response{Message: model.Message{Role: "assistant", ToolCalls: []model.ToolCall{{ID: "read-1", Name: "get_document", Arguments: json.RawMessage(`{"id":"doc-1"}`)}}}}
}
func day03Must(t *testing.T, m model.Model, tools map[string]tool.Tool, n int) *agent.Agent {
	t.Helper()
	a, e := agent.New(m, tools, n)
	if e != nil {
		t.Fatal(e)
	}
	return a
}

type described struct {
	day03Tool
	name string
}

func (d described) Name() string { return d.name }
func (d described) Definition() model.ToolDefinition {
	return model.ToolDefinition{Name: d.name, Parameters: json.RawMessage(`{"type":"object"}`)}
}

type binder struct {
	calls int
	defs  []model.ToolDefinition
	bound model.Model
	err   error
}

func (b *binder) Generate(context.Context, []model.Message) (*model.Response, error) {
	panic("unbound model")
}
func (b *binder) WithTools(d []model.ToolDefinition) (model.Model, error) {
	b.calls++
	b.defs = d
	return b.bound, b.err
}
func TestDay03HostBinding(t *testing.T) {
	b := &binder{bound: day03Model(func(context.Context, []model.Message) (*model.Response, error) { return day03Answer("已读取"), nil })}
	a := day03Must(t, b, map[string]tool.Tool{"z": described{name: "z"}, "a": described{name: "a"}}, 1)
	if b.calls != 1 || len(b.defs) != 2 || b.defs[0].Name != "a" || b.defs[1].Name != "z" {
		t.Fatal(b)
	}
	for i := 0; i < 2; i++ {
		if s, e := a.Run(context.Background(), ""); e != nil || s != "已读取" {
			t.Fatal(s, e)
		}
	}
	if b.calls != 1 {
		t.Fatal("rebound during Run")
	}
	sentinel := errors.New("bind failed")
	b = &binder{err: sentinel}
	if _, e := agent.New(b, nil, 1); !errors.Is(e, sentinel) {
		t.Fatal(e)
	}
	if _, e := agent.New(&binder{}, nil, 1); e == nil {
		t.Fatal("nil bound model")
	}
}

package openai

import (
	"encoding/json"
	"fmt"
	"github.com/linko-zhao/mini-agent-runtime/internal/model"
)

func (a *Adapter) WithTools(defs []model.ToolDefinition) (model.Model, error) {
	copyDefs := make([]model.ToolDefinition, 0, len(defs))
	seen := map[string]bool{}
	for _, d := range defs {
		var schema map[string]any
		if d.Name == "" || seen[d.Name] {
			return nil, fmt.Errorf("openai adapter: empty or duplicate tool name %q", d.Name)
		}
		if err := json.Unmarshal(d.Parameters, &schema); err != nil || schema["type"] != "object" {
			return nil, fmt.Errorf("openai adapter: tool %q requires an object schema", d.Name)
		}
		seen[d.Name] = true
		d.Parameters = append(json.RawMessage(nil), d.Parameters...)
		copyDefs = append(copyDefs, d)
	}
	next := *a
	next.definitions = copyDefs
	return &next, nil
}

package agent

import (
	"fmt"
	"github.com/linko-zhao/mini-agent-runtime/internal/model"
	"github.com/linko-zhao/mini-agent-runtime/internal/tool"
	"sort"
)

func New(m model.Model, tools map[string]tool.Tool, maxSteps int) (*Agent, error) {
	if m == nil || maxSteps <= 0 {
		return nil, fmt.Errorf("agent: model and positive maxSteps required")
	}
	a := &Agent{model: m, tools: make(map[string]tool.Tool), maxSteps: maxSteps}
	names := []string{}
	for n, t := range tools {
		if t == nil || n == "" || n != t.Name() {
			return nil, fmt.Errorf("agent: invalid registration")
		}
		a.tools[n] = t
		names = append(names, n)
	}
	sort.Strings(names)
	var defs []model.ToolDefinition
	for _, n := range names {
		if t, ok := a.tools[n].(tool.DescribedTool); ok {
			d := t.Definition()
			if d.Name != n {
				return nil, fmt.Errorf("agent: definition name mismatch")
			}
			defs = append(defs, d)
		}
	}
	if b, ok := m.(model.ToolBinder); ok {
		bound, err := b.WithTools(defs)
		if err != nil {
			return nil, err
		}
		if bound == nil {
			return nil, fmt.Errorf("agent: nil bound model")
		}
		a.model = bound
	}
	return a, nil
}
