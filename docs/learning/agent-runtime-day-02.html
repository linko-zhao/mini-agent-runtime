今天只解决什么

今天实现 HTTP 模型适配层，让 Runtime 能消费 Provider 的非流式 Chat Completions 响应。

读取 doc-1 时，Provider 的 arguments 是字符串，content 可能为 null，usage 也可能缺失。Loop 若直接解释这些差异，每换 Provider 都会改控制流程。

只做非流式单次 HTTP 请求，不接真实账号、不做自动重试、流式、完整 Provider SDK 或工具说明绑定。测试全部使用注入的 RoundTripper。

先解释概念，再看一次具体运行

适配层

是什么：在两种协议之间翻译的边界。

为什么需要：把 Provider 的 arguments 字符串变成 Runtime 的 JSON 字节。

落到代码：Adapter.Generate 组织 HTTP；encodeRequest/decodeResponse 负责转换。

规范化

是什么：把协议中的可选差异变成稳定输入。

为什么需要：null 正文变空字符串；缺失 usage 变零值。

落到代码：model.Response 保存 Message、FinishReason、Usage。

错误分类

是什么：为调用方提供可识别的失败信息。

为什么需要：429 标记 rate_limited；不直接重新发请求。

落到代码：ModelError 携带 StatusCode、Code、Retryable、Err。

失败反例

HTTP 429 返回 ModelError{Code:rate_limited,Retryable:true}；这只是重试资格提示，本日不自动重试。空 choices、非法 JSON、非 assistant 或坏调用都返回 nil,error。

为什么这样设计

Config / New

职责与字段/参数：Config.APIKey、Model、BaseURL 均要求非空；BaseURL 是完整 endpoint，New 不额外校验 URL 格式。Client 可 nil，回退 http.DefaultClient。

返回与失败：成功返回 Adapter，配置错误返回 nil,error；APIKey 由可信宿主提供，不进入消息。

状态、所有权与取消：配置按值保存，Client 指针共享；调用方配置超时，默认客户端没有总时限。构造不请求网络。

设计理由：HTTP 策略可注入，测试不监听端口。

Adapter.Generate

职责与字段/参数：Generate(ctx,messages) 允许 nil/空消息切片，本页不代替 Provider 做对话完整性校验。POST 到 BaseURL，设置 JSON 和 Bearer 头。

返回与失败：2xx 解码；非 2xx 返回分类错误，不返回部分 Response。传输失败用 ModelError 包装且保留 errors.Is；Code=transport_error、Retryable=true，包括取消，但不自动重试。

状态、所有权与取消：ctx 传到 http.Request；transport 需合作取消。不得关闭调用方 Client；响应 Body 由适配器关闭。无并发承诺。

设计理由：Loop 不需要知道 HTTP 状态、请求头或 Provider 字段。

Response / FinishReason / Usage

职责与字段/参数：Response 新增 FinishReason（字符串，未知值保留）与 Usage。Usage.InputTokens/OutputTokens 是非负场景下的 token 计数，直接映射 Provider，不估算也不额外校验负数。

返回与失败：缺失 usage 是零；缺失 reason 是空；本页提供的稳定 model.go 也含 Day03 的 ToolDefinition/ToolBinder 声明，今天不调用。

状态、所有权与取消：Message/ToolCall 字段继承 Day01；响应只取第一项 choice，不合并多个候选。

设计理由：使调用方能区分协议完成原因与运行循环是否结束。

ModelError.Error / Unwrap

职责与字段/参数：StatusCode 为 HTTP 状态，无状态时零；Code 是分类；Retryable 只是提示；Err 可 nil。Error 有原因时拼 Code 与原因，否则返回 Code；Unwrap 返回 Err。

返回与失败：errors.As 获取分类，errors.Is 检查原因。401 unauthorized/false；429 rate_limited/true；≥500 provider_unavailable/true；其余非2xx provider_error/false。

状态、所有权与取消：没有重试、副作用或状态修改。Provider 私密错误体不回传。

设计理由：保留机器可读错误，避免匹配完整错误文本。

转换函数与 wire 类型

职责与字段/参数：encodeRequest(name,messages,defs) 映射 role/content/tool_calls/tool_call_id；Arguments 必须是合法 JSON，写为 function.arguments 字符串。今天 defs 恒为 nil；wire 结构仅属 Provider 包。

返回与失败：decodeResponse 读取最多 4 MiB+1 字节，超过 4 MiB 拒绝；JSON 完整解码，必须非空 choices、assistant、每个调用 ID/Name 非空且 arguments 合法 JSON；不校验重复 ID 或 function.type。

状态、所有权与取消：无状态；读取失败不吞原错误；null content 和缺失 usage/reason 使用零值。不访问业务资料。

设计理由：明确教学子集，Day09 再检查批次 ID，不把语法校验当业务校验。

type Message struct {

	ToolCalls  []ToolCall

	ID        string
	Name      string
	Arguments json.RawMessage

type FinishReason string

const (
	FinishStop      FinishReason = "stop"
	FinishToolCalls FinishReason = "tool_calls"
	FinishLength    FinishReason = "length"
)

type Usage struct {
	InputTokens  int
	OutputTokens int
}

type Response struct {
	Message      Message
	FinishReason FinishReason
	Usage        Usage
}

type ModelError struct {
	StatusCode int
	Code       string
	Retryable  bool
	Err        error
}

func (e *ModelError) Error() string {
	if e.Err != nil {
		return e.Code + ": " + e.Err.Error()
	}
	return e.Code
}

func (e *ModelError) Unwrap() error {
	return e.Err
}

type Model interface {
	Generate(ctx context.Context, messages []Message) (*Response, error)
}

type ToolDefinition struct {
	Name        string          `json:"name"`
	Description string          `json:"description,omitempty"`
	Parameters  json.RawMessage `json:"parameters,omitempty"`
}

type ToolBinder interface {
	WithTools(defs []ToolDefinition) (Model, error)
}

package openai_test

import (

type transport func(*http.Request) (*http.Response, error)

func (f transport) RoundTrip(r *http.Request) (*http.Response, error) { return f(r) }
func newAdapter(t *testing.T, f transport) *openai.Adapter {
	t.Helper()
	a, e := openai.New(openai.Config{APIKey: "test", Model: "course", BaseURL: "https://provider.invalid/chat", Client: &http.Client{Transport: f}})
	if e != nil {
		t.Fatal(e)
	}
	return a
}
func response(status int, s string) *http.Response {
	return &http.Response{StatusCode: status, Body: io.NopCloser(strings.NewReader(s)), Header: make(http.Header)}
}
func TestDay02Normalize(t *testing.T) {
	a := newAdapter(t, func(r *http.Request) (*http.Response, error) {
		if r.Method != "POST" || r.Header.Get("Authorization") != "Bearer test" || r.Header.Get("Content-Type") != "application/json" {
			t.Fatal("headers")
		}
		var body map[string]any
		if e := json.NewDecoder(r.Body).Decode(&body); e != nil {
			t.Fatal(e)
		}
		ms := body["messages"].([]any)
		if body["model"] != "course" || len(ms) != 1 || ms[0].(map[string]any)["content"] != "读取 doc-1" {
			t.Fatal(body)
		}
		return response(200, `{"choices":[{"message":{"role":"assistant","content":null,"tool_calls":[{"id":"read-1","type":"function","function":{"name":"get_document","arguments":"{\"id\":\"doc-1\"}"}}]},"finish_reason":"future_reason"}],"usage":{"prompt_tokens":3,"completion_tokens":4}}`), nil
	})
	r, e := a.Generate(context.Background(), []model.Message{{Role: "user", Content: "读取 doc-1"}})
	if e != nil {
		t.Fatal(e)
	}
	if r.Message.Role != "assistant" || r.Message.Content != "" || len(r.Message.ToolCalls) != 1 || r.Message.ToolCalls[0].ID != "read-1" || r.Message.ToolCalls[0].Name != "get_document" || string(r.Message.ToolCalls[0].Arguments) != `{"id":"doc-1"}` || r.FinishReason != "future_reason" || r.Usage.InputTokens != 3 || r.Usage.OutputTokens != 4 {
		t.Fatal(r)
	}
	a = newAdapter(t, func(*http.Request) (*http.Response, error) {
		return response(200, `{"choices":[{"message":{"role":"assistant","content":"已读取"}}]}`), nil
	})
	r, e = a.Generate(context.Background(), nil)
	if e != nil || r.Usage != (model.Usage{}) || r.FinishReason != "" {
		t.Fatal(r, e)
	}
}
func TestDay02Failures(t *testing.T) {
	for _, cfg := range []openai.Config{{}, {APIKey: "test"}, {APIKey: "test", Model: "course"}} {
		if _, e := openai.New(cfg); e == nil {
			t.Fatal("config")
		}
	}
	for _, tc := range []struct {
		status int
		code   string
		retry  bool
	}{{401, "unauthorized", false}, {429, "rate_limited", true}, {503, "provider_unavailable", true}, {400, "provider_error", false}} {
		a := newAdapter(t, func(*http.Request) (*http.Response, error) { return response(tc.status, "private provider body"), nil })
		_, e := a.Generate(context.Background(), nil)
		var me *model.ModelError
		if !errors.As(e, &me) || me.StatusCode != tc.status || me.Code != tc.code || me.Retryable != tc.retry {
			t.Fatal(e)
		}
	}
	sentinel := errors.New("offline")
	a := newAdapter(t, func(*http.Request) (*http.Response, error) { return nil, sentinel })
	_, e := a.Generate(context.Background(), nil)
	if !errors.Is(e, sentinel) {
		t.Fatal(e)
	}
	for _, s := range []string{`broken`, `{"choices":[]}`, `{"choices":[{"message":{"role":"user"}}]}`, `{"choices":[{"message":{"role":"assistant","tool_calls":[{"id":"","function":{"name":"get_document","arguments":"{}"}}]}}]}`, `{"choices":[{"message":{"role":"assistant","tool_calls":[{"id":"read-1","function":{"name":"get_document","arguments":"bad"}}]}}]}`, strings.Repeat("x", (4<<20)+1)} {
		a := newAdapter(t, func(*http.Request) (*http.Response, error) { return response(200, s), nil })
		if r, e := a.Generate(context.Background(), nil); e == nil || r != nil {
			t.Fatal("invalid response accepted")
		}
	}
	ctx, cancel := context.WithCancel(context.Background())
	cancel()
	a = newAdapter(t, func(r *http.Request) (*http.Response, error) { return nil, r.Context().Err() })
	_, e = a.Generate(ctx, nil)
	if !errors.Is(e, context.Canceled) {
		t.Fatal(e)
	}
	calls := 0
	a = newAdapter(t, func(*http.Request) (*http.Response, error) { calls++; return response(200, `{}`), nil })
	_, e = a.Generate(context.Background(), []model.Message{{Role: "assistant", ToolCalls: []model.ToolCall{{Arguments: json.RawMessage(`bad`)}}}})
	if e == nil || calls != 0 {
		t.Fatal("invalid request sent")
	}
}

func TestDay02HistoryEncoding(t *testing.T) {
	a := newAdapter(t, func(r *http.Request) (*http.Response, error) {
		var body struct {
			Messages []struct {
				Role    string `json:"role"`
				Content string `json:"content"`
				ID      string `json:"tool_call_id"`
				Calls   []struct {
					ID       string `json:"id"`
					Type     string `json:"type"`
					Function struct {
						Name      string `json:"name"`
						Arguments string `json:"arguments"`
					} `json:"function"`
				} `json:"tool_calls"`
			} `json:"messages"`
		}
		if e := json.NewDecoder(r.Body).Decode(&body); e != nil {
			t.Fatal(e)
		}
		if len(body.Messages) != 3 {
			t.Fatal(body)
		}
		call := body.Messages[1].Calls[0]
		if body.Messages[0].Role != "user" || body.Messages[1].Role != "assistant" || call.ID != "read-1" || call.Type != "function" || call.Function.Name != "get_document" || call.Function.Arguments != `{"id":"doc-1"}` || body.Messages[2].Role != "tool" || body.Messages[2].ID != "read-1" || body.Messages[2].Content != "正文" {
			t.Fatal(body)
		}
		return response(200, `{"choices":[{"message":{"role":"assistant","content":"已读取"}}]}`), nil
	})
	_, e := a.Generate(context.Background(), []model.Message{{Role: "user", Content: "读取 doc-1"}, {Role: "assistant", ToolCalls: []model.ToolCall{{ID: "read-1", Name: "get_document", Arguments: json.RawMessage(`{"id":"doc-1"}`)}}}, {Role: "tool", ToolCallID: "read-1", Content: "正文"}})
	if e != nil {
		t.Fatal(e)
	}
}

// Package openai translates the course Runtime model contract to the
// non-streaming Chat Completions HTTP protocol described on Day 02/03.
package openai

import (
	"bytes"
	"context"
	"fmt"
	"net/http"

	"github.com/linko-zhao/mini-agent-runtime/internal/model"
)

type Config struct {
	APIKey, Model, BaseURL string
	Client                 *http.Client
}
type Adapter struct {
	config      Config
	definitions []model.ToolDefinition
}

func New(config Config) (*Adapter, error) {
	if config.APIKey == "" || config.Model == "" || config.BaseURL == "" {
		return nil, fmt.Errorf("openai adapter: API key, model and base URL are required")
	}
	if config.Client == nil {
		config.Client = http.DefaultClient
	}
	return &Adapter{config: config}, nil
}
func (a *Adapter) Generate(ctx context.Context, messages []model.Message) (*model.Response, error) {
	body, err := encodeRequest(a.config.Model, messages, a.definitions)
	if err != nil {
		return nil, err
	}
	req, err := http.NewRequestWithContext(ctx, http.MethodPost, a.config.BaseURL, bytes.NewReader(body))
	if err != nil {
		return nil, err
	}
	req.Header.Set("Content-Type", "application/json")
	req.Header.Set("Authorization", "Bearer "+a.config.APIKey)
	resp, err := a.config.Client.Do(req)
	if err != nil {
		return nil, &model.ModelError{Code: "transport_error", Retryable: true, Err: err}
	}
	defer resp.Body.Close()
	if resp.StatusCode < 200 || resp.StatusCode >= 300 {
		return nil, httpError(resp.StatusCode)
	}
	return decodeResponse(resp.Body)
}
func httpError(status int) error {
	code, retry := "provider_error", false
	switch {
	case status == 401:
		code = "unauthorized"
	case status == 429:
		code = "rate_limited"
		retry = true
	case status >= 500:
		code = "provider_unavailable"
		retry = true
	}
	return &model.ModelError{StatusCode: status, Code: code, Retryable: retry}
}

package openai

import (
	"bytes"
	"encoding/json"
	"fmt"
	"github.com/linko-zhao/mini-agent-runtime/internal/model"
	"io"
	"net/http"
	"strings"
)

const responseLimit = 4 << 20

// 独立 wire 结构只描述 Provider 协议；Runtime 类型不会被 JSON 直接解码。
type requestEnvelope struct {
	Model    string           `json:"model"`
	Messages []requestMessage `json:"messages"`
	Tools    []requestTool    `json:"tools,omitempty"`
}
type requestMessage struct {
	Role    string        `json:"role"`
	Content string        `json:"content,omitempty"`
	Calls   []requestCall `json:"tool_calls,omitempty"`
	CallID  string        `json:"tool_call_id,omitempty"`
}
type requestCall struct {
	ID       string          `json:"id"`
	Type     string          `json:"type"`
	Function requestFunction `json:"function"`
}
type requestFunction struct {
	Name      string `json:"name"`
	Arguments string `json:"arguments"`
}
type requestTool struct {
	Type     string               `json:"type"`
	Function model.ToolDefinition `json:"function"`
}

func encodeRequest(name string, messages []model.Message, defs []model.ToolDefinition) ([]byte, error) {
	p := requestEnvelope{Model: name}
	for _, m := range messages {
		w := requestMessage{Role: m.Role, Content: m.Content, CallID: m.ToolCallID}
		for _, c := range m.ToolCalls {
			if !json.Valid(c.Arguments) {
				return nil, fmt.Errorf("openai adapter: invalid arguments for call %q", c.ID)
			}
			w.Calls = append(w.Calls, requestCall{ID: c.ID, Type: "function", Function: requestFunction{Name: c.Name, Arguments: string(c.Arguments)}})
		}
		p.Messages = append(p.Messages, w)
	}
	for _, d := range defs {
		p.Tools = append(p.Tools, requestTool{Type: "function", Function: d})
	}
	return json.Marshal(p)
}

type responseEnvelope struct {
	Choices []responseChoice `json:"choices"`
	Usage   *usageWire       `json:"usage"`
}
type responseChoice struct {
	Message responseMessage `json:"message"`
	Reason  *string         `json:"finish_reason"`
}
type responseMessage struct {
	Role    string         `json:"role"`
	Content *string        `json:"content"`
	Calls   []responseCall `json:"tool_calls"`
}
type responseCall struct {
	ID       string          `json:"id"`
	Type     string          `json:"type"`
	Function requestFunction `json:"function"`
}
type usageWire struct {
	Input  int `json:"prompt_tokens"`
	Output int `json:"completion_tokens"`
}

func decodeResponse(r io.Reader) (*model.Response, error) {
	raw, err := io.ReadAll(io.LimitReader(r, responseLimit+1))
	if err != nil {
		return nil, err
	}
	if len(raw) > responseLimit {
		return nil, fmt.Errorf("openai adapter: response exceeds limit")
	}
	var wire responseEnvelope
	if err = json.Unmarshal(raw, &wire); err != nil {
		return nil, fmt.Errorf("openai adapter: invalid response: %w", err)
	}
	if len(wire.Choices) == 0 {
		return nil, fmt.Errorf("openai adapter: response contains no choices")
	}
	c := wire.Choices[0]
	if c.Message.Role != "assistant" {
		return nil, fmt.Errorf("openai adapter: expected assistant response")
	}
	out := &model.Response{Message: model.Message{Role: c.Message.Role}}
	if c.Message.Content != nil {
		out.Message.Content = *c.Message.Content
	}
	for _, call := range c.Message.Calls {
		if call.ID == "" || call.Function.Name == "" || !json.Valid([]byte(call.Function.Arguments)) {
			return nil, fmt.Errorf("openai adapter: malformed tool call")
		}
		out.Message.ToolCalls = append(out.Message.ToolCalls, model.ToolCall{ID: call.ID, Name: call.Function.Name, Arguments: json.RawMessage(strings.TrimSpace(call.Function.Arguments))})
	}
	if c.Reason != nil {
		out.FinishReason = model.FinishReason(*c.Reason)
	}
	if wire.Usage != nil {
		out.Usage = model.Usage{InputTokens: wire.Usage.Input, OutputTokens: wire.Usage.Output}
	}
	return out, nil
}

// statusResponse is handy for protocol-focused tests without a live endpoint.
func statusResponse(status int) *http.Response {
	return &http.Response{StatusCode: status, Body: io.NopCloser(bytes.NewReader(nil))}
}
