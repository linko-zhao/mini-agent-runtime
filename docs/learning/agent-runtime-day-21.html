新增接口为什么放在assembly？

ReplayEvidenceSource

唯一方法Evidence(ctx, StoredCheckpoint)返回ReplayEvidenceSet和error。它代表宿主取得、核验证据的端口，便于替换实现及注入故障。

输入包含Version与Checkpoint；来源必须使用原RunID、版本与调用身份返回证据。它不能把模型自报、readOnly注解或空台账直接当成未开始/后端幂等证明。

适配器应合作取消、支持预期并发，并在调用方处理返回证据时保持缓冲区稳定。测试用内存fake，不实现真实确认服务。

Week03Report

Recovery是Day19 RecoveryPlan，包含位置与完整记录。Replay是Day20 ReplayPlan指针；Ready、Exhausted、Terminal时为nil，ReviewPending成功时非nil。

nil不等于Blocked=false，也不意味着可执行。存在Replay时仍先检查Blocked；未阻塞仅表示给定证据可形成建议。

Recovery.Record、Replay.Record、Replay.Steps和来源持有的输入相互隔离。报告包含正文与参数，不是公开事件。

ReviewWeek03(ctx, store, source, runID)

返回Week03Report与error；先委托PlanRecovery处理ctx、store、key和读取校验。只Load一次，无重试。

非ReviewPending直接返回Recovery，Replay=nil，完全不查证据，source可nil。ReviewPending要求非nil有效source，否则返回ErrEvidenceSourceRequired。

调用Evidence前复制记录；用原ctx调用一次。返回后先检查ctx.Err，再处理来源错误，最后委托PlanReplay；所有失败返回零报告，依赖错误保留errors.Is/As。

Run管当前状态更新，Store管保存，Source管事实来源，PlanReplay管依据。把这些宿主选择塞进Run，会混淆生命周期和执行策略。

R5 · 只读与副本隔离

审查不Save、不Apply，不增加Step、Version或改变历史。来源修改自己的记录输入，不污染Recovery；调用方改报告，也不污染存储、来源或报告的另一字段。

故障矩阵：先预测，再看断言

正常与阻塞

Ready/预算/终态：Load=1、Evidence=0、Replay=nil。Pending：Load=1、Evidence=1，按原依据给出Blocked或未阻塞计划。

所有场景审查期间Save=0，Run与存储前后不变。Unknown是有效的阻塞结论，不是网络失败。

失败与取消

Store失败/未找到/非法记录：不查来源。来源失败/缺失/旧证据：零报告，保留可识别原因；没有自动重试。

预取消：两种依赖均0次；加载中取消：不进入来源；来源中取消：拒收迟到数据。nilctx、nilstore、空白key沿前序规则拒绝。

// Package runstate models in-memory transitions. It is not a checkpoint store.
package runstate

import (
	"errors"
	"strings"

	"github.com/linko-zhao/mini-agent-runtime/internal/model"
)

type Phase string

const (
	Ready        Phase = "ready"
	ModelRunning Phase = "model_running"
	ToolsPending Phase = "tools_pending"
	ToolsRunning Phase = "tools_running"
	Completed    Phase = "completed"
	Failed       Phase = "failed"
	Cancelled    Phase = "cancelled"
)

type EventKind string

const (
	StartModel    EventKind = "start_model"
	ModelReturned EventKind = "model_returned"
	StartTools    EventKind = "start_tools"
	ToolsReturned EventKind = "tools_returned"
	Abort         EventKind = "abort"
	Cancel        EventKind = "cancel"
)

var (
	ErrTransition = errors.New("runstate: invalid transition")
	ErrMaxSteps   = errors.New("runstate: max model rounds exceeded")
)

type State struct {
	RunID           string
	Phase           Phase
	Step, MaxSteps  int
	Messages        []model.Message
	Pending         []model.ToolCall
	Result, Failure string
}
type Event struct {
	Kind     EventKind
	Response *model.Response
	Results  []model.Message
	Reason   string
}

func New(runID, input string, maxSteps int) (State, error) {
	if strings.TrimSpace(runID) == "" || maxSteps <= 0 {
		return State{}, ErrTransition
	}
	return State{RunID: runID, Phase: Ready, MaxSteps: maxSteps, Messages: []model.Message{{Role: "user", Content: input}}}, nil
}
func cloneCalls(calls []model.ToolCall) []model.ToolCall {
	if calls == nil {
		return nil
	}
	out := append([]model.ToolCall{}, calls...)
	for i := range out {
		out[i].Arguments = append([]byte(nil), out[i].Arguments...)
	}
	return out
}
func cloneMessages(ms []model.Message) []model.Message {
	if ms == nil {
		return nil
	}
	out := append([]model.Message{}, ms...)
	for i := range out {
		out[i].ToolCalls = cloneCalls(out[i].ToolCalls)
	}
	return out
}
func cloneState(s State) State {
	s.Messages = cloneMessages(s.Messages)
	s.Pending = cloneCalls(s.Pending)
	return s
}
func terminal(p Phase) bool { return p == Completed || p == Failed || p == Cancelled }

// Advance has no I/O. The host performs calls between start/returned events.
// It accepts states from New/Advance and never mutates input state/event buffers.
func Advance(current State, e Event) (State, error) {
	if terminal(current.Phase) {
		return State{}, ErrTransition
	}
	next := cloneState(current)
	switch e.Kind {
	case StartModel:
		if current.Phase != Ready {
			return State{}, ErrTransition
		}
		if current.Step >= current.MaxSteps {
			return State{}, ErrMaxSteps
		}
		next.Step++
		next.Phase = ModelRunning
	case ModelReturned:
		if current.Phase != ModelRunning || e.Response == nil || e.Response.Message.Role != "assistant" {
			return State{}, ErrTransition
		}
		msg := e.Response.Message
		ids := map[string]bool{}
		for _, call := range msg.ToolCalls {
			if call.ID == "" || strings.TrimSpace(call.Name) == "" || ids[call.ID] {
				return State{}, ErrTransition
			}
			ids[call.ID] = true
		}
		next.Messages = append(next.Messages, cloneMessages([]model.Message{msg})...)
		if len(msg.ToolCalls) == 0 {
			next.Phase = Completed
			next.Result = msg.Content
		} else {
			next.Phase = ToolsPending
			next.Pending = cloneCalls(msg.ToolCalls)
		}
	case StartTools:
		if current.Phase != ToolsPending {
			return State{}, ErrTransition
		}
		next.Phase = ToolsRunning
	case ToolsReturned:
		if current.Phase != ToolsRunning || len(e.Results) != len(current.Pending) {
			return State{}, ErrTransition
		}
		for i, msg := range e.Results {
			if msg.Role != "tool" || msg.ToolCallID != current.Pending[i].ID {
				return State{}, ErrTransition
			}
		}
		next.Messages = append(next.Messages, cloneMessages(e.Results)...)
		next.Pending = nil
		next.Phase = Ready
	case Abort:
		if strings.TrimSpace(e.Reason) == "" {
			return State{}, ErrTransition
		}
		next.Phase = Failed
		next.Failure = e.Reason
	case Cancel:
		next.Phase = Cancelled
		next.Failure = "cancelled"
	default:
		return State{}, ErrTransition
	}
	return next, nil
}

internal/runstate/run.go

package runstate

import (
	"context"
	"sync"
)

// Run owns one in-memory State. Use its pointer; do not copy it after use.
// The zero value is not usable; construct with NewRun.
type Run struct {
	ctx   context.Context
	mu    sync.Mutex
	state State
}

// NewRun retains ctx for cancellation/deadlines, not for storing State.
// A cancelled context rejects construction; invalid identity/budget uses ErrTransition.
func NewRun(ctx context.Context, runID, input string, maxSteps int) (*Run, error) {
	if ctx == nil {
		return nil, ErrTransition
	}
	if err := ctx.Err(); err != nil {
		return nil, err
	}
	s, err := New(runID, input, maxSteps)
	if err != nil {
		return nil, err
	}
	return &Run{ctx: ctx, state: s}, nil
}

// Context returns the original context. The host owns any cancel function.
// Model/tool I/O must use this context outside Apply, never while holding its lock.
func (r *Run) Context() context.Context { return r.ctx }

// Snapshot returns an independent copy without triggering transitions.
func (r *Run) Snapshot() State {
	r.mu.Lock()
	defer r.mu.Unlock()
	return cloneState(r.state)
}

// Apply commits exactly one valid event, or returns the unchanged current copy.
// For a nonterminal Run, observed context cancellation commits Cancel and rejects
// the event with the original context error. A terminal State is never overwritten.
// This serializes transitions, not a snapshot -> external I/O -> Apply transaction.
func (r *Run) Apply(e Event) (State, error) {
	r.mu.Lock()
	defer r.mu.Unlock()
	if terminal(r.state.Phase) {
		return cloneState(r.state), ErrTransition
	}
	if err := r.ctx.Err(); err != nil {
		next, transitionErr := Advance(r.state, Event{Kind: Cancel})
		if transitionErr != nil {
			return cloneState(r.state), transitionErr
		}
		r.state = next
		return cloneState(r.state), err
	}
	next, err := Advance(r.state, e)
	if err != nil {
		return cloneState(r.state), err
	}
	r.state = next
	return cloneState(r.state), nil
}

internal/runstate/checkpoint.go

package runstate

import "errors"

type CheckpointBoundary string

const (
	CheckpointReady       CheckpointBoundary = "ready"
	CheckpointBeforeTools CheckpointBoundary = "before_tools"
	CheckpointTerminal    CheckpointBoundary = "terminal"
)

var ErrCheckpointBoundary = errors.New("runstate: checkpoint requires a stable boundary")

// Checkpoint is an in-memory candidate, not a storage acknowledgement or resume token.
// State uses the existing message/call contract. Boundary classifies State.Phase.
type Checkpoint struct {
	Boundary CheckpointBoundary
	State    State
}

// Clone copies nested mutable state. It does not validate or authorize a checkpoint.
func (c Checkpoint) Clone() Checkpoint {
	c.State = cloneState(c.State)
	return c
}

// Checkpoint samples boundary and state under the same lock.
// It does not observe context cancellation, advance state, pause the driver,
// perform I/O, or promise that an external tool has not already taken effect.
func (r *Run) Checkpoint() (Checkpoint, error) {
	r.mu.Lock()
	defer r.mu.Unlock()
	var boundary CheckpointBoundary
	switch r.state.Phase {
	case Ready:
		boundary = CheckpointReady
	case ToolsPending:
		boundary = CheckpointBeforeTools
	case Completed, Failed, Cancelled:
		boundary = CheckpointTerminal
	default:
		return Checkpoint{}, ErrCheckpointBoundary
	}
	return Checkpoint{Boundary: boundary, State: cloneState(r.state)}, nil
}

internal/runstate/store.go

package runstate

import (
	"context"
	"errors"
	"strings"
	"sync"
)

var (
	ErrCheckpointNotFound = errors.New("runstate: checkpoint not found")
	ErrCheckpointConflict = errors.New("runstate: checkpoint version conflict")
	ErrInvalidCheckpoint  = errors.New("runstate: invalid checkpoint request")
)

// StoredCheckpoint is the latest saved value. Version is a per-RunID CAS token,
// not State.Step, a schema version, an execution lease or a resume permission.
type StoredCheckpoint struct {
	Version    uint64
	Checkpoint Checkpoint
}

type CheckpointStore interface {
	Load(context.Context, string) (StoredCheckpoint, error)
	Save(context.Context, Checkpoint, uint64) (StoredCheckpoint, error)
}

// MemoryCheckpointStore is process-local and not durable. Its zero value works.
// Do not copy after use. It retains only the latest record for each exact RunID.
type MemoryCheckpointStore struct {
	mu      sync.Mutex
	records map[string]StoredCheckpoint
}

func NewMemoryCheckpointStore() *MemoryCheckpointStore { return &MemoryCheckpointStore{} }

func cloneStored(v StoredCheckpoint) StoredCheckpoint {
	v.Checkpoint = v.Checkpoint.Clone()
	return v
}
func storeContext(ctx context.Context) error {
	if ctx == nil {
		return ErrInvalidCheckpoint
	}
	return ctx.Err()
}

// validateCandidate checks only the trusted candidate envelope, not full message
// history, authorization, authenticity or an untrusted serialized payload.
func validateCandidate(cp Checkpoint) bool {
	s := cp.State
	if strings.TrimSpace(s.RunID) == "" || s.MaxSteps <= 0 || s.Step < 0 || s.Step > s.MaxSteps {
		return false
	}
	switch cp.Boundary {
	case CheckpointReady:
		return s.Phase == Ready
	case CheckpointBeforeTools:
		return s.Phase == ToolsPending
	case CheckpointTerminal:
		return terminal(s.Phase)
	default:
		return false
	}
}

func (s *MemoryCheckpointStore) Load(ctx context.Context, runID string) (StoredCheckpoint, error) {
	if err := storeContext(ctx); err != nil {
		return StoredCheckpoint{}, err
	}
	if strings.TrimSpace(runID) == "" {
		return StoredCheckpoint{}, ErrInvalidCheckpoint
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	if err := ctx.Err(); err != nil {
		return StoredCheckpoint{}, err
	}
	v, ok := s.records[runID]
	if !ok {
		return StoredCheckpoint{}, ErrCheckpointNotFound
	}
	return cloneStored(v), nil
}

// Save: expected=0 creates only if absent; otherwise it must equal current Version.
// A successful write increments Version even for identical content.
// Errors in this in-memory implementation return zero and leave stored data intact.
// Cancellation is sampled before and after locking; later cancellation can still
// accompany a committed success. The lock wait itself is not context-interruptible.
func (s *MemoryCheckpointStore) Save(ctx context.Context, cp Checkpoint, expected uint64) (StoredCheckpoint, error) {
	if err := storeContext(ctx); err != nil {
		return StoredCheckpoint{}, err
	}
	if expected == ^uint64(0) || !validateCandidate(cp) {
		return StoredCheckpoint{}, ErrInvalidCheckpoint
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	if err := ctx.Err(); err != nil {
		return StoredCheckpoint{}, err
	}
	current := s.records[cp.State.RunID]
	if current.Version != expected {
		return StoredCheckpoint{}, ErrCheckpointConflict
	}
	next := StoredCheckpoint{Version: expected + 1, Checkpoint: cp.Clone()}
	if s.records == nil {
		s.records = make(map[string]StoredCheckpoint)
	}
	s.records[cp.State.RunID] = next
	return cloneStored(next), nil
}

package runstate

import (
	"bytes"
	"context"
	"strings"
)

type RecoveryPosition string

const (
	RecoveryReady         RecoveryPosition = "recorded_ready"
	RecoveryReviewPending RecoveryPosition = "review_pending"
	RecoveryTerminal      RecoveryPosition = "terminal"
	RecoveryExhausted     RecoveryPosition = "budget_exhausted"
)

// RecoveryPlan is a read-only description of one loaded record, not permission
// to execute or a restored Run. Record retains the store's observed Version.
type RecoveryPlan struct {
	Position RecoveryPosition
	Record   StoredCheckpoint
}

// recoveryRecordValid performs limited consistency checks, not complete replay,
// provenance verification, schema migration or external-effect reconciliation.
func recoveryRecordValid(v StoredCheckpoint, runID string) bool {
	cp := v.Checkpoint
	s := cp.State
	if v.Version == 0 || s.RunID != runID || !validateCandidate(cp) || len(s.Messages) == 0 || s.Messages[0].Role != "user" {
		return false
	}
	last := s.Messages[len(s.Messages)-1]
	if s.Phase == Completed {
		return len(s.Pending) == 0 && last.Role == "assistant" && len(last.ToolCalls) == 0 && s.Result == last.Content
	}
	if s.Phase == Ready {
		return len(s.Pending) == 0 && len(last.ToolCalls) == 0
	}
	if s.Phase == ToolsPending && len(s.Pending) == 0 {
		return false
	}
	if len(s.Pending) > 0 {
		if last.Role != "assistant" || len(last.ToolCalls) != len(s.Pending) {
			return false
		}
		ids := map[string]bool{}
		for i, c := range s.Pending {
			h := last.ToolCalls[i]
			if c.ID == "" || strings.TrimSpace(c.Name) == "" || ids[c.ID] || c.ID != h.ID || c.Name != h.Name || !bytes.Equal(c.Arguments, h.Arguments) {
				return false
			}
			ids[c.ID] = true
		}
	}
	return true
}

// PlanRecovery loads exactly once; never Save, Apply, Generate or Invoke.
// Pre-cancelled contexts cause no Load. Cancellation observed after Load wins over
// any returned value/error. Later cancellation can still follow a successful read.
// The injected store must support the caller's concurrency and owned return data.
func PlanRecovery(ctx context.Context, store CheckpointStore, runID string) (RecoveryPlan, error) {
	if err := storeContext(ctx); err != nil {
		return RecoveryPlan{}, err
	}
	if store == nil || strings.TrimSpace(runID) == "" {
		return RecoveryPlan{}, ErrInvalidCheckpoint
	}
	v, err := store.Load(ctx, runID)
	if cancelErr := ctx.Err(); cancelErr != nil {
		return RecoveryPlan{}, cancelErr
	}
	if err != nil {
		return RecoveryPlan{}, err
	}
	if !recoveryRecordValid(v, runID) {
		return RecoveryPlan{}, ErrInvalidCheckpoint
	}
	position := RecoveryTerminal
	switch v.Checkpoint.State.Phase {
	case Ready:
		position = RecoveryReady
		if v.Checkpoint.State.Step >= v.Checkpoint.State.MaxSteps {
			position = RecoveryExhausted
		}
	case ToolsPending:
		position = RecoveryReviewPending
	}
	return RecoveryPlan{Position: position, Record: cloneStored(v)}, nil
}

package runstate

import (
	"bytes"
	"context"
	"errors"
	"strings"

	"github.com/linko-zhao/mini-agent-runtime/internal/model"
)

type ReplayBasis string

const (
	BasisUnknown      ReplayBasis = "unknown"
	BasisNotStarted   ReplayBasis = "not_started"
	BasisResultKnown  ReplayBasis = "result_known"
	BasisBackendDedup ReplayBasis = "backend_dedup"
)

type ReplayAction string

const (
	ActionHold   ReplayAction = "hold"
	ActionInvoke ReplayAction = "invoke"
	ActionReuse  ReplayAction = "reuse_result"
)

var ErrReplayEvidence = errors.New("runstate: invalid replay evidence")

// Evidence is a trusted host assertion, not model output or a verified backend
// receipt. BackendDedup means the host has separately verified key scope,
// payload binding, validity window and same-operation replay behavior.
type ReplayEvidence struct {
	Call           model.ToolCall
	Basis          ReplayBasis
	Result         *model.Message
	IdempotencyKey string
}
type ReplayEvidenceSet struct {
	RunID   string
	Version uint64
	Calls   []ReplayEvidence
}
type ReplayStep struct {
	Call           model.ToolCall
	Action         ReplayAction
	Result         *model.Message
	IdempotencyKey string
}
type ReplayPlan struct {
	Record  StoredCheckpoint
	Steps   []ReplayStep
	Blocked bool // any Hold blocks the entire batch before effects/history updates
}

// PlanReplay is pure decision logic. It never loads/saves, invokes tools or
// grants a lease. Even an unblocked plan needs fresh ownership/authorization.
// Inputs must remain immutable while it runs; returned nested data is owned.
func PlanReplay(ctx context.Context, recovery RecoveryPlan, evidence ReplayEvidenceSet) (ReplayPlan, error) {
	if err := storeContext(ctx); err != nil {
		return ReplayPlan{}, err
	}
	record := recovery.Record
	if recovery.Position != RecoveryReviewPending || record.Checkpoint.State.Phase != ToolsPending || !recoveryRecordValid(record, record.Checkpoint.State.RunID) {
		return ReplayPlan{}, ErrInvalidCheckpoint
	}
	calls := record.Checkpoint.State.Pending
	if evidence.RunID != record.Checkpoint.State.RunID || evidence.Version != record.Version || len(evidence.Calls) != len(calls) {
		return ReplayPlan{}, ErrReplayEvidence
	}
	out := ReplayPlan{Record: cloneStored(record), Steps: make([]ReplayStep, len(calls))}
	keys := map[string]bool{}
	for i, c := range calls {
		if err := ctx.Err(); err != nil {
			return ReplayPlan{}, err
		}
		e := evidence.Calls[i]
		if e.Call.ID != c.ID || e.Call.Name != c.Name || !bytes.Equal(e.Call.Arguments, c.Arguments) {
			return ReplayPlan{}, ErrReplayEvidence
		}
		step := ReplayStep{Call: cloneCalls([]model.ToolCall{c})[0]}
		switch e.Basis {
		case BasisUnknown:
			if e.Result != nil || e.IdempotencyKey != "" {
				return ReplayPlan{}, ErrReplayEvidence
			}
			step.Action = ActionHold
			out.Blocked = true
		case BasisNotStarted:
			if e.Result != nil || e.IdempotencyKey != "" {
				return ReplayPlan{}, ErrReplayEvidence
			}
			step.Action = ActionInvoke
		case BasisResultKnown:
			if e.Result == nil || e.Result.Role != "tool" || e.Result.ToolCallID != c.ID || len(e.Result.ToolCalls) != 0 || e.IdempotencyKey != "" {
				return ReplayPlan{}, ErrReplayEvidence
			}
			msg := cloneMessages([]model.Message{*e.Result})[0]
			step.Action = ActionReuse
			step.Result = &msg
		case BasisBackendDedup:
			if e.Result != nil || strings.TrimSpace(e.IdempotencyKey) == "" || keys[e.IdempotencyKey] {
				return ReplayPlan{}, ErrReplayEvidence
			}
			keys[e.IdempotencyKey] = true
			step.Action = ActionInvoke
			step.IdempotencyKey = e.IdempotencyKey
		default:
			return ReplayPlan{}, ErrReplayEvidence
		}
		out.Steps[i] = step
	}
	if err := ctx.Err(); err != nil {
		return ReplayPlan{}, err
	}
	return out, nil
}

package assembly

import (
	"context"
	"errors"

	"github.com/linko-zhao/mini-agent-runtime/internal/runstate"
)

// ReplayEvidenceSource is a trusted host adapter, not a model/tool assertion.
// It must cooperate with context and support the intended concurrency.
type ReplayEvidenceSource interface {
	Evidence(context.Context, runstate.StoredCheckpoint) (runstate.ReplayEvidenceSet, error)
}

var ErrEvidenceSourceRequired = errors.New("assembly: pending review requires an evidence source")

type Week03Report struct {
	Recovery runstate.RecoveryPlan
	Replay   *runstate.ReplayPlan // nil for Ready, Exhausted or Terminal
}

// ReviewWeek03 composes the Week03 read-only path; it does not restore or execute.
// Exactly one Load is delegated to PlanRecovery. Only ReviewPending consults the
// source once, using a separate record copy, then delegates to PlanReplay.
// Every failure returns a zero report; dependency errors preserve their chains.
func ReviewWeek03(ctx context.Context, store runstate.CheckpointStore, source ReplayEvidenceSource, runID string) (Week03Report, error) {
	recovery, err := runstate.PlanRecovery(ctx, store, runID)
	if err != nil {
		return Week03Report{}, err
	}
	if err = ctx.Err(); err != nil {
		return Week03Report{}, err
	}
	if recovery.Position != runstate.RecoveryReviewPending {
		return Week03Report{Recovery: recovery}, nil
	}
	if source == nil {
		return Week03Report{}, ErrEvidenceSourceRequired
	}
	input := recovery.Record
	input.Checkpoint = input.Checkpoint.Clone()
	evidence, err := source.Evidence(ctx, input)
	if cancelErr := ctx.Err(); cancelErr != nil {
		return Week03Report{}, cancelErr
	}
	if err != nil {
		return Week03Report{}, err
	}
	replay, err := runstate.PlanReplay(ctx, recovery, evidence)
	if err != nil {
		return Week03Report{}, err
	}
	return Week03Report{Recovery: recovery, Replay: &replay}, nil
}
