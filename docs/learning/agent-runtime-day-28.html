接口卡片：宿主组合契约

ToolBatchExecutor

func(context.Context, []model.ToolCall) ([]model.Message, error) 是真实业务I/O端口的宿主回调。接收原Run.Context和独立Pending副本，要求合作取消、按原CallID与顺序返回工具消息。可修改本次输入副本；返回缓冲区不得在调用方处理时并发改写。

不是结果查询接口，也不是本页实现的具体工具。宿主应复用已建立的授权、策略与工具适配；本日不自动构建Registry/Agent或绕过权限。任何error都不自动重试；同次部分results不回填。

Week04Host / BuildWeek04

BuildWeek04(ctx, runID, input, maxSteps, execute) (*Week04Host,error) 先委托NewRun验证上下文、身份和轮次预算，再要求非nil执行回调，否则返回nil / ErrWeek04ExecutorRequired；随后构造唯一Gate。原NewRun错误保持，例如nilctx的ErrTransition、已取消ctx的原错误。

私有run保存原执行ctx与状态，gate保存唯一门禁，execute保存回调。零值不可用，使用后不复制；本日不接受外部配对的Run/Gate，避免误用另一条运行的ctx。

Gate()

返回同一个*PauseGate用于宿主模型事件、Pause和状态读取。它不会生成模型响应或自动开始运行。Continue期间普通推进由Continue独占，其它写入只限取消；禁止并发Continue或绕过门禁。

Week04Report

  • Pause：已完成捕获/解码阶段的PauseSnapshot；早期失败可能为零，编码失败时可能保留已捕获候选。
  • Payload []byte：已完成编码阶段的Gob数据，未编码/编码失败为nil，不是已保存、已恢复或已授权凭据。
  • State：每次返回前重新取得的独立当前State，包括错误返回；可为ToolsPending、ToolsRunning、Ready、Failed或Cancelled。
  • ExecutionErr：执行端口实际返回的原错误，即使后续核验成功也保留；没有执行错误则nil。
  • Reconciled bool：只有走了执行错误分支并且Reconcile成功才true；正常执行成功是false。

报告保留已完成阶段，错误时并非零报告契约；与Day26的零报告不同。Pause、Payload和State分别拥有数据，不与Host串改；错误对象保持原原因链，来源需遵守输出所有权。

Continue(decision, queryCtx, source) (Week04Report,error)

  1. 起点是当前持有暂停，先Decide；错误立即返回当前State，不执行后续。
  2. PauseSnapshot → Encode → Decode。失败保留已完成的报告字段，不消费、不回滚已记录决定、不保存；重发同一决定仍受原ErrDecisionRecorded约束。
  3. ResolveDecision按目标消费。reject得到Failed、nil即返回，不执行、不查询；批准得到ToolsRunning。
  4. 执行前观察原ctx；取消则通过Gate.Apply记录取消并返回原错误。否则调用执行回调恰好一次，传原Run.Context。
  5. 保存ExecutionErr，执行后再观察原ctx；取消优先，丢弃迟到results/错误，不查询，不回滚已发生外部操作。
  6. 执行无error时用原ToolsReturned回填；错误结果形状由旧状态机拒绝，不回退成自动核验。
  7. 执行有error且原ctx仍有效，才调用ReconcileToolResults(queryCtx,source)。它的错误原样作为本次error，执行错误仍在Report.ExecutionErr中；成功则Reconciled=true。

queryCtx/source只在执行错误分支需要，正常执行或拒绝时可为nil；缺来源、查询取消、Unknown/NotStarted均沿Day26/27返回，不重执行业务操作。最终State总是返回前独立快照；只有本次error为nil且State符合预期才进入后续阶段，拒绝nilerror的Failed也不是继续模型信号。

ctx在边界采样，最后检查后仍可取消；回调须合作取消，不强杀协程。若取消在最后检查后被Run.Apply观察，仍按原终态规则处理；不存在物理时间优先或副作用回滚承诺。

// Package runstate models in-memory transitions. It is not a checkpoint store.
package runstate

import (
	"errors"
	"strings"

	"github.com/linko-zhao/mini-agent-runtime/internal/model"
)

type Phase string

const (
	Ready        Phase = "ready"
	ModelRunning Phase = "model_running"
	ToolsPending Phase = "tools_pending"
	ToolsRunning Phase = "tools_running"
	Completed    Phase = "completed"
	Failed       Phase = "failed"
	Cancelled    Phase = "cancelled"
)

type EventKind string

const (
	StartModel    EventKind = "start_model"
	ModelReturned EventKind = "model_returned"
	StartTools    EventKind = "start_tools"
	ToolsReturned EventKind = "tools_returned"
	Abort         EventKind = "abort"
	Cancel        EventKind = "cancel"
)

var (
	ErrTransition = errors.New("runstate: invalid transition")
	ErrMaxSteps   = errors.New("runstate: max model rounds exceeded")
)

type State struct {
	RunID           string
	Phase           Phase
	Step, MaxSteps  int
	Messages        []model.Message
	Pending         []model.ToolCall
	Result, Failure string
}
type Event struct {
	Kind     EventKind
	Response *model.Response
	Results  []model.Message
	Reason   string
}

func New(runID, input string, maxSteps int) (State, error) {
	if strings.TrimSpace(runID) == "" || maxSteps <= 0 {
		return State{}, ErrTransition
	}
	return State{RunID: runID, Phase: Ready, MaxSteps: maxSteps, Messages: []model.Message{{Role: "user", Content: input}}}, nil
}
func cloneCalls(calls []model.ToolCall) []model.ToolCall {
	if calls == nil {
		return nil
	}
	out := append([]model.ToolCall{}, calls...)
	for i := range out {
		out[i].Arguments = append([]byte(nil), out[i].Arguments...)
	}
	return out
}
func cloneMessages(ms []model.Message) []model.Message {
	if ms == nil {
		return nil
	}
	out := append([]model.Message{}, ms...)
	for i := range out {
		out[i].ToolCalls = cloneCalls(out[i].ToolCalls)
	}
	return out
}
func cloneState(s State) State {
	s.Messages = cloneMessages(s.Messages)
	s.Pending = cloneCalls(s.Pending)
	return s
}
func terminal(p Phase) bool { return p == Completed || p == Failed || p == Cancelled }

// Advance has no I/O. The host performs calls between start/returned events.
// It accepts states from New/Advance and never mutates input state/event buffers.
func Advance(current State, e Event) (State, error) {
	if terminal(current.Phase) {
		return State{}, ErrTransition
	}
	next := cloneState(current)
	switch e.Kind {
	case StartModel:
		if current.Phase != Ready {
			return State{}, ErrTransition
		}
		if current.Step >= current.MaxSteps {
			return State{}, ErrMaxSteps
		}
		next.Step++
		next.Phase = ModelRunning
	case ModelReturned:
		if current.Phase != ModelRunning || e.Response == nil || e.Response.Message.Role != "assistant" {
			return State{}, ErrTransition
		}
		msg := e.Response.Message
		ids := map[string]bool{}
		for _, call := range msg.ToolCalls {
			if call.ID == "" || strings.TrimSpace(call.Name) == "" || ids[call.ID] {
				return State{}, ErrTransition
			}
			ids[call.ID] = true
		}
		next.Messages = append(next.Messages, cloneMessages([]model.Message{msg})...)
		if len(msg.ToolCalls) == 0 {
			next.Phase = Completed
			next.Result = msg.Content
		} else {
			next.Phase = ToolsPending
			next.Pending = cloneCalls(msg.ToolCalls)
		}
	case StartTools:
		if current.Phase != ToolsPending {
			return State{}, ErrTransition
		}
		next.Phase = ToolsRunning
	case ToolsReturned:
		if current.Phase != ToolsRunning || len(e.Results) != len(current.Pending) {
			return State{}, ErrTransition
		}
		for i, msg := range e.Results {
			if msg.Role != "tool" || msg.ToolCallID != current.Pending[i].ID {
				return State{}, ErrTransition
			}
		}
		next.Messages = append(next.Messages, cloneMessages(e.Results)...)
		next.Pending = nil
		next.Phase = Ready
	case Abort:
		if strings.TrimSpace(e.Reason) == "" {
			return State{}, ErrTransition
		}
		next.Phase = Failed
		next.Failure = e.Reason
	case Cancel:
		next.Phase = Cancelled
		next.Failure = "cancelled"
	default:
		return State{}, ErrTransition
	}
	return next, nil
}

package runstate

import (
	"context"
	"sync"
)

// Run owns one in-memory State. Use its pointer; do not copy it after use.
// The zero value is not usable; construct with NewRun.
type Run struct {
	ctx   context.Context
	mu    sync.Mutex
	state State
}

// NewRun retains ctx for cancellation/deadlines, not for storing State.
// A cancelled context rejects construction; invalid identity/budget uses ErrTransition.
func NewRun(ctx context.Context, runID, input string, maxSteps int) (*Run, error) {
	if ctx == nil {
		return nil, ErrTransition
	}
	if err := ctx.Err(); err != nil {
		return nil, err
	}
	s, err := New(runID, input, maxSteps)
	if err != nil {
		return nil, err
	}
	return &Run{ctx: ctx, state: s}, nil
}

// Context returns the original context. The host owns any cancel function.
// Model/tool I/O must use this context outside Apply, never while holding its lock.
func (r *Run) Context() context.Context { return r.ctx }

// Snapshot returns an independent copy without triggering transitions.
func (r *Run) Snapshot() State {
	r.mu.Lock()
	defer r.mu.Unlock()
	return cloneState(r.state)
}

// Apply commits exactly one valid event, or returns the unchanged current copy.
// For a nonterminal Run, observed context cancellation commits Cancel and rejects
// the event with the original context error. A terminal State is never overwritten.
// This serializes transitions, not a snapshot -> external I/O -> Apply transaction.
func (r *Run) Apply(e Event) (State, error) {
	r.mu.Lock()
	defer r.mu.Unlock()
	if terminal(r.state.Phase) {
		return cloneState(r.state), ErrTransition
	}
	if err := r.ctx.Err(); err != nil {
		next, transitionErr := Advance(r.state, Event{Kind: Cancel})
		if transitionErr != nil {
			return cloneState(r.state), transitionErr
		}
		r.state = next
		return cloneState(r.state), err
	}
	next, err := Advance(r.state, e)
	if err != nil {
		return cloneState(r.state), err
	}
	r.state = next
	return cloneState(r.state), nil
}

// Checkpoint is an in-memory candidate, not a storage acknowledgement or resume token.
// State uses the existing message/call contract. Boundary classifies State.Phase.
type Checkpoint struct {
	Boundary CheckpointBoundary
	State    State
}

// Clone copies nested mutable state. It does not validate or authorize a checkpoint.
func (c Checkpoint) Clone() Checkpoint {
	c.State = cloneState(c.State)
	return c
}

// Checkpoint samples boundary and state under the same lock.
// It does not observe context cancellation, advance state, pause the driver,
// perform I/O, or promise that an external tool has not already taken effect.
func (r *Run) Checkpoint() (Checkpoint, error) {
	r.mu.Lock()
	defer r.mu.Unlock()
	var boundary CheckpointBoundary
	switch r.state.Phase {
	case Ready:
		boundary = CheckpointReady
	case ToolsPending:
		boundary = CheckpointBeforeTools
	case Completed, Failed, Cancelled:
		boundary = CheckpointTerminal
	default:
		return Checkpoint{}, ErrCheckpointBoundary
	}
	return Checkpoint{Boundary: boundary, State: cloneState(r.state)}, nil
}

func NewMemoryCheckpointStore() *MemoryCheckpointStore { return &MemoryCheckpointStore{} }

func cloneStored(v StoredCheckpoint) StoredCheckpoint {
	v.Checkpoint = v.Checkpoint.Clone()
	return v

func storeContext(ctx context.Context) error {
	if ctx == nil {
		return ErrInvalidCheckpoint
	}
	return ctx.Err()
}

// validateCandidate checks only the trusted candidate envelope, not full message
// history, authorization, authenticity or an untrusted serialized payload.
func validateCandidate(cp Checkpoint) bool {
	s := cp.State
	if strings.TrimSpace(s.RunID) == "" || s.MaxSteps <= 0 || s.Step < 0 || s.Step > s.MaxSteps {
		return false
	}
	switch cp.Boundary {
	case CheckpointReady:
		return s.Phase == Ready
	case CheckpointBeforeTools:
		return s.Phase == ToolsPending
	case CheckpointTerminal:
		return terminal(s.Phase)
	default:
		return false
	}
}

func (s *MemoryCheckpointStore) Load(ctx context.Context, runID string) (StoredCheckpoint, error) {
	if err := storeContext(ctx); err != nil {
		return StoredCheckpoint{}, err
	}
	if strings.TrimSpace(runID) == "" {
		return StoredCheckpoint{}, ErrInvalidCheckpoint
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	if err := ctx.Err(); err != nil {
		return StoredCheckpoint{}, err
	}
	v, ok := s.records[runID]
	if !ok {
		return StoredCheckpoint{}, ErrCheckpointNotFound
	}
	return cloneStored(v), nil
}

// Save: expected=0 creates only if absent; otherwise it must equal current Version.
// A successful write increments Version even for identical content.
// Errors in this in-memory implementation return zero and leave stored data intact.
// Cancellation is sampled before and after locking; later cancellation can still
// accompany a committed success. The lock wait itself is not context-interruptible.
func (s *MemoryCheckpointStore) Save(ctx context.Context, cp Checkpoint, expected uint64) (StoredCheckpoint, error) {
	if err := storeContext(ctx); err != nil {
		return StoredCheckpoint{}, err
	}
	if expected == ^uint64(0) || !validateCandidate(cp) {
		return StoredCheckpoint{}, ErrInvalidCheckpoint
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	if err := ctx.Err(); err != nil {
		return StoredCheckpoint{}, err
	}
	current := s.records[cp.State.RunID]
	if current.Version != expected {
		return StoredCheckpoint{}, ErrCheckpointConflict
	}
	next := StoredCheckpoint{Version: expected + 1, Checkpoint: cp.Clone()}
	if s.records == nil {
		s.records = make(map[string]StoredCheckpoint)
	}
	s.records[cp.State.RunID] = next
	return cloneStored(next), nil
}

		return len(s.Pending) == 0 && len(last.ToolCalls) == 0
	}
	if s.Phase == ToolsPending && len(s.Pending) == 0 {
		return false

			return false

// grants a lease. Even an unblocked plan needs fresh ownership/authorization.
// Inputs must remain immutable while it runs; returned nested data is owned.
func PlanReplay(ctx context.Context, recovery RecoveryPlan, evidence ReplayEvidenceSet) (ReplayPlan, error) {

		return ReplayPlan{}, err

		return ReplayPlan{}, ErrInvalidCheckpoint

		return ReplayPlan{}, ErrReplayEvidence
	}
	out := ReplayPlan{Record: cloneStored(record), Steps: make([]ReplayStep, len(calls))}
	keys := map[string]bool{}
	for i, c := range calls {
		if err := ctx.Err(); err != nil {
			return ReplayPlan{}, err
		}
		e := evidence.Calls[i]
		if e.Call.ID != c.ID || e.Call.Name != c.Name || !bytes.Equal(e.Call.Arguments, c.Arguments) {
			return ReplayPlan{}, ErrReplayEvidence
		}
		step := ReplayStep{Call: cloneCalls([]model.ToolCall{c})[0]}
		switch e.Basis {
		case BasisUnknown:
			if e.Result != nil || e.IdempotencyKey != "" {
				return ReplayPlan{}, ErrReplayEvidence
			}
			step.Action = ActionHold
			out.Blocked = true
		case BasisNotStarted:
			if e.Result != nil || e.IdempotencyKey != "" {
				return ReplayPlan{}, ErrReplayEvidence
			}
			step.Action = ActionInvoke
		case BasisResultKnown:
			if e.Result == nil || e.Result.Role != "tool" || e.Result.ToolCallID != c.ID || len(e.Result.ToolCalls) != 0 || e.IdempotencyKey != "" {
				return ReplayPlan{}, ErrReplayEvidence
			}
			msg := cloneMessages([]model.Message{*e.Result})[0]
			step.Action = ActionReuse
			step.Result = &msg
		case BasisBackendDedup:
			if e.Result != nil || strings.TrimSpace(e.IdempotencyKey) == "" || keys[e.IdempotencyKey] {

			step.IdempotencyKey = e.IdempotencyKey
		default:
			return ReplayPlan{}, ErrReplayEvidence
		}
		out.Steps[i] = step
	}
	if err := ctx.Err(); err != nil {
		return ReplayPlan{}, err
	}
	return out, nil
}

// It samples the original Run context and performs no I/O or state transition.

	g.mu.Lock()
	defer g.mu.Unlock()
	if err := g.run.Context().Err(); err != nil {
		return PauseRequest{}, err
	}
	if strings.TrimSpace(id) == "" || strings.TrimSpace(reason) == "" {
		return PauseRequest{}, ErrInvalidPause
	}
	if g.request != nil {
		return PauseRequest{}, ErrPaused
	}
	cp, err := g.run.Checkpoint()
	if err != nil || cp.State.Phase != ToolsPending {
		return PauseRequest{}, ErrPauseBoundary
	}
	if err := g.run.Context().Err(); err != nil {
		return PauseRequest{}, err
	}
	request := PauseRequest{ID: id, Reason: reason, Checkpoint: cp}
	g.request = &request
	out := request
	out.Checkpoint = out.Checkpoint.Clone()
	return out, nil
}

// event is rejected with a current owned State and ErrPaused. Observed context
// cancellation is allowed through to Run.Apply; terminal state clears the pause.
// The host may perform I/O only AFTER a successful transition; no I/O runs here.

	g.mu.Lock()
	defer g.mu.Unlock()
	if g.request != nil && g.run.Context().Err() == nil {
		return g.run.Snapshot(), ErrPaused
	}
	state, err := g.run.Apply(e)
	if terminal(state.Phase) {
		g.request = nil
		g.decision = nil
	}
	return state, err
}

	st := cp.State
	if s.SchemaVersion != PauseSnapshotSchema || strings.TrimSpace(p.ID) == "" || strings.TrimSpace(p.Reason) == "" ||

		return false
	}
	ids := map[string]bool{}
	for i, call := range st.Pending {
		h := last.ToolCalls[i]
		if call.ID == "" || strings.TrimSpace(call.Name) == "" || ids[call.ID] || call.ID != h.ID || call.Name != h.Name || !bytes.Equal(call.Arguments, h.Arguments) {
			return false

			return false

// No I/O/ctx is involved. This is not canonical encoding or a storage commit.

	last := s.Messages[len(s.Messages)-1]
	if last.Role != "assistant" || len(last.ToolCalls) != len(s.Pending) {
		return false
	}
	ids := map[string]bool{}
	for i, call := range s.Pending {
		h := last.ToolCalls[i]
		if call.ID == "" || strings.TrimSpace(call.Name) == "" || ids[call.ID] || call.ID != h.ID || call.Name != h.Name || !bytes.Equal(call.Arguments, h.Arguments) {
			return false
		}
		ids[call.ID] = true
	}
	return true
}

// InspectToolOutcomes queries each pending logical call once in input order.
// Every failure returns zero report. It never Apply/Save/Invoke, retries a
// query or converts missing responses into confirmed not-started evidence.
func InspectToolOutcomes(ctx context.Context, source ToolOutcomeSource, observed State) (ToolOutcomeReport, error) {
	if ctx == nil {
		return ToolOutcomeReport{}, ErrOutcomeState
	}
	if err := ctx.Err(); err != nil {
		return ToolOutcomeReport{}, err
	}
	if !outcomeStateValid(observed) {
		return ToolOutcomeReport{}, ErrOutcomeState
	}
	if source == nil {
		return ToolOutcomeReport{}, ErrOutcomeSourceRequired
	}
	out := ToolOutcomeReport{Observed: cloneState(observed), Evidence: make([]ToolOutcomeEvidence, len(observed.Pending)), AllResultsKnown: true}
	results := make([]model.Message, len(observed.Pending))
	for i, call := range out.Observed.Pending {
		if err := ctx.Err(); err != nil {
			return ToolOutcomeReport{}, err
		}
		input := ToolAttempt{RunID: observed.RunID, Step: observed.Step, Call: cloneCalls([]model.ToolCall{call})[0]}
		e, err := source.Lookup(ctx, input)
		if cancelErr := ctx.Err(); cancelErr != nil {
			return ToolOutcomeReport{}, cancelErr
		}
		if err != nil {
			return ToolOutcomeReport{}, err
		}
		v := e.Evidence
		if e.RunID != observed.RunID || e.Step != observed.Step || v.Call.ID != call.ID || v.Call.Name != call.Name ||
			!bytes.Equal(v.Call.Arguments, call.Arguments) || v.IdempotencyKey != "" {
			return ToolOutcomeReport{}, ErrReplayEvidence
		}
		switch v.Basis {
		case BasisUnknown, BasisNotStarted:
			if v.Result != nil {
				return ToolOutcomeReport{}, ErrReplayEvidence
			}
			out.AllResultsKnown = false
		case BasisResultKnown:
			if v.Result == nil || v.Result.Role != "tool" || v.Result.ToolCallID != call.ID || len(v.Result.ToolCalls) != 0 {
				return ToolOutcomeReport{}, ErrReplayEvidence
			}
			results[i] = cloneMessages([]model.Message{*v.Result})[0]
			msg := cloneMessages([]model.Message{*v.Result})[0]
			v.Result = &msg
		default: // BackendDedup is a retry policy assertion, not a known outcome.
			return ToolOutcomeReport{}, ErrReplayEvidence
		}
		v.Call = cloneCalls([]model.ToolCall{call})[0]
		out.Evidence[i] = ToolOutcomeEvidence{RunID: e.RunID, Step: e.Step, Evidence: v}
	}
	if err := ctx.Err(); err != nil {
		return ToolOutcomeReport{}, err
	}
	if out.AllResultsKnown {
		out.Results = results
	}
	return out, nil
}

var (
	ErrOutcomeIncomplete = errors.New("runstate: not all tool results are known")
	ErrOutcomeConflict   = errors.New("runstate: run changed during outcome inspection")

// ReconcileToolResults repairs missing tool messages on THIS live Run only.
// Queries occur outside the gate lock. The original observation is compared
// and ToolsReturned committed under one gate lock, preventing duplicate commits
// through this unique gate. It never executes/retries business tools or restores
// a process. Every return includes an owned current State; callers must check err.
func (g *PauseGate) ReconcileToolResults(queryCtx context.Context, source ToolOutcomeSource) (State, error) {
	observed := g.Snapshot()
	report, err := InspectToolOutcomes(queryCtx, source, observed)
	if err != nil {
		return g.Snapshot(), err
	}
	if !report.AllResultsKnown {
		return g.Snapshot(), ErrOutcomeIncomplete
	}
	g.mu.Lock()
	defer g.mu.Unlock()
	current := g.run.Snapshot()
	if err := queryCtx.Err(); err != nil {
		return current, err
	}
	if !reflect.DeepEqual(current, report.Observed) {
		return current, ErrOutcomeConflict
	}
	// Run.Apply observes the ORIGINAL execution ctx. A fresh query ctx cannot
	// revive a cancelled execution, and a late query cancellation cannot roll back
	// a successful commit. Do not recursively call Gate.Apply while holding g.mu.
	state, err := g.run.Apply(Event{Kind: ToolsReturned, Results: report.Results})
	if terminal(state.Phase) {
		g.request, g.decision = nil, nil
	}
	return state, err
}

	"github.com/linko-zhao/mini-agent-runtime/internal/runstate"
)

var ErrWeek04ExecutorRequired = errors.New("assembly: week04 executor required")

// ToolBatchExecutor is the host's actual I/O boundary, not an outcome lookup.
// It must cooperate with ctx and return results in original call order.
type ToolBatchExecutor func(context.Context, []model.ToolCall) ([]model.Message, error)

type Week04Host struct {
	run     *runstate.Run
	gate    *runstate.PauseGate
	execute ToolBatchExecutor
}

// Week04Report retains the completed teaching stages even on error. Payload is
// encoded data, NOT a saved receipt. ExecutionErr is retained after reconciliation.
type Week04Report struct {
	Pause        runstate.PauseSnapshot
	Payload      []byte
	State        runstate.State
	ExecutionErr error
	Reconciled   bool
}

func BuildWeek04(ctx context.Context, runID, input string, maxSteps int, execute ToolBatchExecutor) (*Week04Host, error) {
	r, err := runstate.NewRun(ctx, runID, input, maxSteps)
	if err != nil {
		return nil, err
	}
	if execute == nil {
		return nil, ErrWeek04ExecutorRequired
	}
	g, err := runstate.NewPauseGate(r)
	if err != nil {
		return nil, err
	}
	return &Week04Host{run: r, gate: g, execute: execute}, nil
}

// Gate returns the SAME gate for host-driven model events and pause requests.
// Continue owns normal advancement while it runs; other host writes are limited
// to cancellation. Do not concurrently run Continue or bypass this gate.
func (h *Week04Host) Gate() *runstate.PauseGate { return h.gate }

// Continue composes recording, codec, resolution, execution and optional outcome
// reconciliation. It starts from a held pause, never retries tools and never saves
// or recreates a Run. A valid queryCtx/source is needed only after execution error.
func (h *Week04Host) Continue(decision runstate.ApprovalDecision, queryCtx context.Context, source runstate.ToolOutcomeSource) (report Week04Report, err error) {
	defer func() { report.State = h.gate.Snapshot() }()
	if _, err = h.gate.Decide(decision); err != nil {
		return
	}
	report.Pause, err = h.gate.PauseSnapshot()
	if err != nil {
		return
	}
	report.Payload, err = runstate.EncodePauseSnapshot(report.Pause)
	if err != nil {
		return
	}
	report.Pause, err = runstate.DecodePauseSnapshot(report.Payload)
	if err != nil {
		return
	}
	state, resolveErr := h.gate.ResolveDecision(decision.RunID, decision.RequestID)
	if resolveErr != nil {
		err = resolveErr
		return
	}
	if state.Phase == runstate.Failed { // a host rejection is successful handling
		return
	}
	if h.run.Context().Err() != nil {
		_, err = h.gate.Apply(runstate.Event{Kind: runstate.Cancel})
		return
	}
	// state is an owned copy; the execution port may modify its input buffers.
	results, executionErr := h.execute(h.run.Context(), state.Pending)
	report.ExecutionErr = executionErr
	if h.run.Context().Err() != nil {
		_, err = h.gate.Apply(runstate.Event{Kind: runstate.Cancel})
		return
	}
	if executionErr == nil {
		_, err = h.gate.Apply(runstate.Event{Kind: runstate.ToolsReturned, Results: results})
		return
	}
	_, err = h.gate.ReconcileToolResults(queryCtx, source)
	report.Reconciled = err == nil
	return
}
