今天只解决什么？

实现 WithPolicy 包装器，让已有工具统一遵守超时、重试资格和输出字节预算。

读取 doc-1 时后端短暂失败，随后恢复；同样的错误若发生在写入工具，重试却可能重复产生副作用。

把每种工具都改成自带重试，容易遗漏资格检查。包装器接收同一个 Execute 接口，将运行策略放在业务外面。

不实现写入去重、持久幂等或跨 Run 恢复。ReadOnly 由可信宿主配置，不信任模型或远端自报。

先把概念说清楚

包装器：原接口外加一层

外部仍调用 Execute；内部先处理策略，再调用原工具。

Name 与 Definition 保持原样，Loop 无需改变。

逻辑调用与尝试

模型的一条 ToolCall 是一次逻辑调用。

内部可能尝试两次，但只向模型回填一条结果，调用 ID 不变。

字节预算：限制返回成本

MaxResultBytes 按 UTF-8 字节计数。

中文通常不止一个字节；超限报受控错误，不截断可能是 JSON 的结果。

失败时停在哪里？

输出“中文”是 6 个 UTF-8 字节。预算为 5 时返回 ExecutionError 和空结果，不把截断文本或原结果送给模型；后端已执行的动作不会被撤销。

这些字段、接口和方法为什么存在？

Policy：六项显式配置

Timeout 是整次逻辑调用的总时限；MaxAttempts 是包含首次在内的上限，0 归一为 1。

RetryDelay 是两次尝试的等待时间。

只读资格和分类器

ReadOnly 表示宿主确认可只读重试；Retryable(error) 判断未知后端错误是否暂时性。

MaxAttempts>1 时两者缺一就构造失败；只读不等于被授权。

结果预算与零值

MaxResultBytes=0 不限；正值限制成功输出。

Timeout/RetryDelay=0 分别表示不额外限时/不等待；这些数值及 MaxAttempts 均不接受负值。

policyTool 保存什么？

target 保存原 DescribedTool，policy 保存值拷贝。

构造不执行目标；Name、Definition 委托目标，不复制其业务状态。

注入目标和 Retryable 的并发安全由宿主保证。

WithPolicy 的输入输出

输入非 nil 的 DescribedTool 和合法 Policy，返回同一接口的新对象。

共享注册表可逐项包装后交给原 Agent.New。

Execute 的取消与拷贝

入场先检查取消；整次调用只派生一次超时。

每次尝试独立复制 Arguments；执行后若 ctx 已取消，优先返回取消。

为什么不重试所有 error？

tool.Error 已表达可回填的业务问题；Canceled/DeadlineExceeded 表示应停止。

只有其他错误被 Retryable 接受、且还有次数，才等待并重试。

R3 · 有限的只读重试

R4 · 共用截止时间

R5 · 成功结果预算

成功字符串长度按字节比较，等于预算允许，大于预算返回空字符串与 tool.ExecutionError。错误路径不把原成功文本带回。

package tool

import (
	"context"
	"encoding/json"
	"errors"
	"github.com/linko-zhao/mini-agent-runtime/internal/model"
	"sync"
	"testing"
	"time"
)

type policyFake struct {
	fn func(context.Context, json.RawMessage) (string, error)
}

func (f policyFake) Name() string                     { return "lookup" }
func (f policyFake) Definition() model.ToolDefinition { return model.ToolDefinition{Name: "lookup"} }
func (f policyFake) Execute(ctx context.Context, a json.RawMessage) (string, error) {
	return f.fn(ctx, a)
}

func TestPolicyReadRetryAndBudget(t *testing.T) {
	transient := errors.New("transient")
	calls := 0
	wrapped, err := WithPolicy(policyFake{func(_ context.Context, a json.RawMessage) (string, error) {
		calls++
		if string(a) != "{}" {
			t.Fatalf("mutated args: %s", a)
		}
		a[0] = 'x'
		if calls == 1 {
			return "", transient
		}
		return "ok", nil
	}}, Policy{MaxAttempts: 2, ReadOnly: true, Retryable: func(e error) bool { return errors.Is(e, transient) }, MaxResultBytes: 2})
	if err != nil {
		t.Fatal(err)
	}
	out, err := wrapped.Execute(context.Background(), json.RawMessage("{}"))
	if err != nil || out != "ok" || calls != 2 || wrapped.Name() != "lookup" || wrapped.Definition().Name != "lookup" {
		t.Fatalf("%q %v calls=%d", out, err, calls)
	}
	budget, _ := WithPolicy(policyFake{func(context.Context, json.RawMessage) (string, error) { return "中文", nil }}, Policy{MaxResultBytes: 5})
	out, err = budget.Execute(context.Background(), nil)
	var known *Error
	if out != "" || !errors.As(err, &known) || known.Kind != ExecutionError {
		t.Fatalf("%q %v", out, err)
	}
}

func TestPolicyNoUnsafeRetry(t *testing.T) {
	for _, p := range []Policy{{MaxAttempts: 2}, {MaxAttempts: 2, ReadOnly: true}, {Timeout: -1}, {RetryDelay: -1}, {MaxResultBytes: -1}} {
		if _, err := WithPolicy(policyFake{}, p); err == nil {
			t.Fatalf("accepted %+v", p)
		}
	}
	for _, failure := range []error{&Error{Kind: ValidationError, Message: "bad input"}, context.Canceled, context.DeadlineExceeded, errors.New("permanent")} {
		calls := 0
		wrapped, _ := WithPolicy(policyFake{func(context.Context, json.RawMessage) (string, error) { calls++; return "", failure }}, Policy{MaxAttempts: 3, ReadOnly: true, Retryable: func(e error) bool { return e != failure || failure.Error() != "permanent" }})
		_, err := wrapped.Execute(context.Background(), nil)
		if !errors.Is(err, failure) || calls != 1 {
			t.Fatalf("%v calls=%d", err, calls)
		}
	}
}

func TestPolicyCancellationAndTimeout(t *testing.T) {
	calls := 0
	wrapped, _ := WithPolicy(policyFake{func(ctx context.Context, _ json.RawMessage) (string, error) {
		calls++
		<-ctx.Done()
		return "late", nil
	}}, Policy{Timeout: 5 * time.Millisecond})
	_, err := wrapped.Execute(context.Background(), nil)
	if !errors.Is(err, context.DeadlineExceeded) || calls != 1 {
		t.Fatalf("%v calls=%d", err, calls)
	}
	ctx, cancel := context.WithCancel(context.Background())
	cancel()
	_, err = wrapped.Execute(ctx, nil)
	if !errors.Is(err, context.Canceled) || calls != 1 {
		t.Fatalf("%v calls=%d", err, calls)
	}
}

func TestPolicyCancellationDuringRetryWait(t *testing.T) {
	parent, cancel := context.WithCancel(context.Background())
	defer cancel()
	ctx := &policyWaitContext{Context: parent, entered: make(chan struct{})}
	go func() { <-ctx.entered; cancel() }()
	calls := 0
	failure := errors.New("temporary")
	wrapped, _ := WithPolicy(policyFake{func(context.Context, json.RawMessage) (string, error) {
		calls++
		return "", failure
	}}, Policy{MaxAttempts: 3, ReadOnly: true, Retryable: func(error) bool { return true }, RetryDelay: time.Hour})
	_, err := wrapped.Execute(ctx, nil)
	if !errors.Is(err, context.Canceled) || calls != 1 {
		t.Fatalf("%v calls=%d", err, calls)
	}
}

func TestPolicyDefaultNoRetry(t *testing.T) {
	calls := 0
	cause := errors.New("transient")
	wrapped, _ := WithPolicy(policyFake{func(context.Context, json.RawMessage) (string, error) { calls++; return "", cause }}, Policy{})
	_, err := wrapped.Execute(context.Background(), nil)
	if !errors.Is(err, cause) || calls != 1 {
		t.Fatalf("%v calls=%d", err, calls)
	}
}

// Done 被重试等待观察到后才取消，避免用 sleep 猜时序。
type policyWaitContext struct {
	context.Context
	entered chan struct{}
	once    sync.Once
}

func (c *policyWaitContext) Done() <-chan struct{} {
	c.once.Do(func() { close(c.entered) })
	return c.Context.Done()
}

func TestDay10AttemptLimitAndSharedDeadline(t *testing.T) {
	failure := errors.New("temporary")
	n := 0
	var deadline time.Time
	wrapped, err := WithPolicy(policyFake{func(ctx context.Context, _ json.RawMessage) (string, error) {
		n++
		d, ok := ctx.Deadline()
		if !ok {
			t.Fatal("missing deadline")
		}
		if n == 1 {
			deadline = d
		} else if !d.Equal(deadline) {
			t.Fatal("timeout restarted between attempts")
		}
		return "", failure
	}}, Policy{Timeout: time.Hour, ReadOnly: true, MaxAttempts: 2, Retryable: func(error) bool { return true }})
	if err != nil {
		t.Fatal(err)
	}
	_, err = wrapped.Execute(context.Background(), nil)
	if !errors.Is(err, failure) || n != 2 {
		t.Fatalf("%v attempts=%d", err, n)
	}
}

func TestDay10ParentDeadline(t *testing.T) {
	ctx, cancel := context.WithTimeout(context.Background(), time.Millisecond)
	defer cancel()
	target := policyFake{func(c context.Context, _ json.RawMessage) (string, error) { <-c.Done(); return "late", nil }}
	wrapped, err := WithPolicy(target, Policy{Timeout: time.Hour})
	if err != nil {
		t.Fatal(err)
	}
	out, err := wrapped.Execute(ctx, nil)
	if out != "" || !errors.Is(err, context.DeadlineExceeded) {
		t.Fatalf("%q %v", out, err)
	}
}

为什么内部两次尝试只产生一条 tool 消息？

Loop 看到的是同一次 Execute；尝试属于包装器内部，不新增模型调用 ID。

ReadOnly 能代替权限判断吗？

不能。

为什么超限不直接截断？

输出可能是结构化 JSON；截断会破坏协议，也不能回滚外部操作。

package tool

import (
	"context"
	"encoding/json"
	"errors"
	"fmt"
	"time"

	"github.com/linko-zhao/mini-agent-runtime/internal/model"
)

// Policy applies to the whole logical call, including all retries and waits.
// ReadOnly is an explicit application assertion, never inferred from a model.
type Policy struct {
	Timeout        time.Duration
	MaxAttempts    int
	ReadOnly       bool
	Retryable      func(error) bool
	RetryDelay     time.Duration
	MaxResultBytes int
}

// WithPolicy preserves the legacy described-tool boundary used by Agent.
// No automatic retry occurs unless both ReadOnly and Retryable permit it.
func WithPolicy(t DescribedTool, p Policy) (DescribedTool, error) {
	if t == nil || p.Timeout < 0 || p.MaxAttempts < 0 || p.RetryDelay < 0 || p.MaxResultBytes < 0 {
		return nil, fmt.Errorf("tool: invalid policy")
	}
	if p.MaxAttempts == 0 {
		p.MaxAttempts = 1
	}
	if p.MaxAttempts > 1 && (!p.ReadOnly || p.Retryable == nil) {
		return nil, fmt.Errorf("tool: retries require explicit read-only and retryable policy")
	}
	return &policyTool{target: t, policy: p}, nil
}

type policyTool struct {
	target DescribedTool
	policy Policy
}

func (t *policyTool) Name() string                     { return t.target.Name() }
func (t *policyTool) Definition() model.ToolDefinition { return t.target.Definition() }
func (t *policyTool) Execute(ctx context.Context, args json.RawMessage) (string, error) {
	if err := ctx.Err(); err != nil {
		return "", err
	}
	if t.policy.Timeout > 0 {
		var cancel context.CancelFunc
		ctx, cancel = context.WithTimeout(ctx, t.policy.Timeout)
		defer cancel()
	}
	for attempt := 1; attempt <= t.policy.MaxAttempts; attempt++ {
		if err := ctx.Err(); err != nil {
			return "", err
		}
		result, err := t.target.Execute(ctx, append(json.RawMessage(nil), args...))
		if ctx.Err() != nil {
			return "", ctx.Err()
		}
		if err == nil {
			if t.policy.MaxResultBytes > 0 && len(result) > t.policy.MaxResultBytes {
				return "", &Error{Kind: ExecutionError, Message: "tool result exceeds byte budget"}
			}
			return result, nil
		}
		if errors.Is(err, context.Canceled) || errors.Is(err, context.DeadlineExceeded) {
			return "", err
		}
		var controlled *Error
		if errors.As(err, &controlled) || attempt == t.policy.MaxAttempts || t.policy.Retryable == nil || !t.policy.Retryable(err) {
			return "", err
		}
		if t.policy.RetryDelay > 0 {
			timer := time.NewTimer(t.policy.RetryDelay)
			select {
			case <-ctx.Done():
				timer.Stop()
				return "", ctx.Err()
			case <-timer.C:
			}
		}
	}
	panic("unreachable")
}
